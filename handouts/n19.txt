19.1 树状数组倒着走
Build	the	answer	bit	by	bit;	O(log	n).
IDEA Walking	down	a	Fenwick	tree ( 树状数组倒着走  / 树状数组上二分 ): to find the smallest position p
with prefix sum ≥ k, do not binary search with O(log n) queries each (O(log² n)). Instead build p bit by bit
from the top: for step = 2⌊log n⌋ down to 1, if p + step ≤ n and t[p + step] < k, move p += step and k −= t[p].
The answer is p + 1. O(log n).
KEY POINT It works because t[p + step] covers exactly (p, p + step] when p is a multiple of 2·step, which holds at
every stage of the walk. With a Fenwick tree of value counts this is "k-th smallest in a multiset" in O(log
n), the job usually given to a balanced tree.
竞 赛 拓 展 专 题  · NOI 篇  · NOI 专 题
Find the k-th element by walking down a Fenwick tree, turn subtrees into ranges with DFS order, and

fenwick_kth.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 const int V = 100000;
5 int t[V + 1];
6 int LOG;
7  
8 void add(int i, int v) { for (; i <= V; i += i & -i) t[i] += v; }
9 int kth(int k) {                                 // smallest x with count(<= x) >= k
10     int p = 0;
11     for (int step = 1 << LOG; step > 0; step >>= 1)
12         if (p + step <= V && t[p + step] < k) { p += step; k -= t[p]; }
13     return p + 1;
14 }
15  
16 int main() {
17     LOG = 0; while ((1 << (LOG + 1)) <= V) LOG++;
18     int q; cin >> q;
19     while (q--) {
20         int op, x; cin >> op >> x;
21         if (op == 1) add(x, 1);                  // insert value x
22         else cout << kth(x) << "\n";            // x-th smallest
23     }
24     return 0;
25 }

IDEA DFS	order (DFS 序 ): number vertices by entry time tin[u] and record tout[u] = the last entry time inside
u's subtree. Then the subtree of u is exactly the range [tin[u], tout[u]], so subtree operations become range
operations on an array (Fenwick or segment tree).
KEY POINT Combinations with two Fenwick trees: subtree add + single-vertex query (range add, point query); vertex
add + subtree sum (point add, range sum); subtree add + subtree sum (range add, range sum, 拓展篇  第 26
章 ). Path-to-root sums: add v at tin[u] and −v at tout[u] + 1, then a point query at tin[x] sums the values of
x's ancestors (Promotion Counting P3605 uses DFS order with a Fenwick tree over values).

dfs_order.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 const int N = 100005;
5 int n, tin[N], tout[N], timer_ = 0;
6 vector<int> g[N];
7 long long A[N], B[N];                            // two Fenwick trees for range add / range sum
8  
9 void dfs(int u, int p) {
10     tin[u] = ++timer_;
11     for (int k = 0; k < int(g[u].size()); k++) if (g[u][k] != p) dfs(g[u][k], u);
12     tout[u] = timer_;
13 }
14 void upd(long long *t, int i, long long v) { for (; i <= n; i += i & -i) t[i] += v; }
15 long long qry(long long *t, int i) { long long s = 0; for (; i > 0; i -= i & -i) s += t[i]; return s; }
16 void rangeAdd(int l, int r, long long v) { upd(A, l, v); upd(A, r + 1, -v); upd(B, l, v * l); upd(B, r + 
1, -v * (r + 1)); }
17 long long prefix(int x) { return qry(A, x) * (x + 1) - qry(B, x); }
18 long long rangeSum(int l, int r) { return prefix(r) - prefix(l - 1); }
19  
20 int main() {
21     int q;
22     cin >> n >> q;
23     for (int i = 2; i <= n; i++) { int p; cin >> p; g[p].push_back(i); g[i].push_back(p); }
24     dfs(1, 0);
25     while (q--) {
26         int op, u; cin >> op >> u;
27         if (op == 1) { long long x; cin >> x; rangeAdd(tin[u], tout[u], x); }     // whole subtree
28         else if (op == 3) { long long x; cin >> x; rangeAdd(tin[u], tin[u], x); } // one vertex
29         else if (op == 2) cout << rangeSum(tin[u], tin[u]) << "\n";              // one vertex
30         else cout << rangeSum(tin[u], tout[u]) << "\n";                          // whole subtree
31     }
32     return 0;
33 }

19.3 可持久化 	Trie
Prefix	versions;	count	differences.
IDEA A persistent	0-1	trie ( 可持久化  Trie): version i = the trie of a1 … ai, built from version i − 1 by copying
one root-to-leaf path (like a persistent segment tree, 进阶篇  第 8 章 ), with a count in every node. Counts of
version r minus version l − 1 describe exactly al … ar.
KEY POINT Maximum of x ⊕ ai over i ∈ [l, r]: walk both versions from the top bit, prefer the opposite bit of x
whenever the count difference there is positive. With prefix XORs it answers "maximum XOR of a suffix
ending in a range" ( 最大异或和  P4735).
persistent_trie.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 const int N = 100005, BITS = 6, M = N * (BITS + 2);
5 int ch[M][2], cnt[M], root[N], tot = 0;
6  
7 int insert(int pre, int x) {                    // copy the path of x
8     int rt = ++tot, cur = rt;
9     for (int b = BITS - 1; b >= 0; b--) {
10         int c = (x >> b) & 1;
11         ch[cur][0] = ch[pre][0]; ch[cur][1] = ch[pre][1]; cnt[cur] = cnt[pre] + 1;
12         ch[cur][c] = ++tot;
13         cur = ch[cur][c]; pre = ch[pre][c];
14     }
15     cnt[cur] = cnt[pre] + 1;
16     return rt;
17 }
18 int query(int u, int v, int x) {                // max x ^ a over versions v minus u
19     int res = 0;
20     for (int b = BITS - 1; b >= 0; b--) {
21         int c = (x >> b) & 1;
22         if (cnt[ch[v][c ^ 1]] - cnt[ch[u][c ^ 1]] > 0) { res |= 1 << b; u = ch[u][c ^ 1]; v = ch[v][c ^ 
1]; }
23         else { u = ch[u][c]; v = ch[v][c]; }
24     }
25     return res;
26 }
27  
28 int main() {
29     int n, q;
30     cin >> n >> q;
31     for (int i = 1; i <= n; i++) { int a; cin >> a; root[i] = insert(root[i - 1], a); }
32     while (q--) { int l, r, x; cin >> l >> r >> x; cout << query(root[l - 1], root[r], x) << "\n"; }
33     return 0;
34 }

19.4 前缀线性基
Keep	the	newest	vector	at	each	bit.
IDEA A prefix	linear	basis ( 前缀线性基 ) answers "maximum xor of a subset of a[l..r]" online. Keep, for every
prefix r, a basis where each vector remembers the largest	position it comes from. When inserting ar with
position r, at each bit where the stored vector is older, swap them (the newer one stays, the older one
continues down).
KEY POINT Then the basis of prefix r restricted to vectors with position ≥ l spans exactly the xors of a[l..r]. Query:
greedy from the high bit, using only vectors with pos ≥ l. O(log V) per insert and query (CF1100F Ivan and
Burgers).

prefix_basis.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 const int N = 500005, LOG = 20;
5 int val[N][LOG], pos[N][LOG];
6  
7 int main() {
8     int n, q;
9     cin >> n;
10     for (int i = 1; i <= n; i++) {
11         int x, p = i;
12         cin >> x;
13         for (int b = 0; b < LOG; b++) { val[i][b] = val[i - 1][b]; pos[i][b] = pos[i - 1][b]; }
14         for (int b = LOG - 1; b >= 0; b--) {
15             if (!(x >> b & 1)) continue;
16             if (!val[i][b]) { val[i][b] = x; pos[i][b] = p; break; }
17             if (pos[i][b] < p) { swap(val[i][b], x); swap(pos[i][b], p); }   // keep the newer vector
18             x ^= val[i][b];
19         }
20     }
21     cin >> q;
22     for (int k = 0; k < q; k++) {
23         int l, r, res = 0;
24         cin >> l >> r;
25         for (int b = LOG - 1; b >= 0; b--)
26             if (pos[r][b] >= l && (res ^ val[r][b]) > res) res ^= val[r][b];
27         cout << res << "\n";
28     }
29     return 0;
30 }

19.5 线段树上二分
One	walk	down	the	tree;	O(log	n).
IDEA Binary	search	on	a	segment	tree ( 线段树上二分 ): to find the first index i ≥ l with ai ≥ x, do not binary
search outside with O(log n) queries each (O(log² n)). Walk the tree once: skip any node outside [l, n] or
with max < x; otherwise go left first, then right. O(log n), because only O(log n) nodes are partly inside and
each full node is entered only if its answer is there.
KEY POINT The same walk finds the k-th one (descend by counts), the first prefix with sum ≥ s (descend by sums, 第 1
节  is the Fenwick version), and the last index ≤ r satisfying a condition (go right first).

1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 const int N = 200005;
5 int n, mx[4 * N];
6  
7 void update(int o, int l, int r, int i, int x) {
8     if (l == r) { mx[o] = x; return; }
9     int m = (l + r) / 2;
10     if (i <= m) update(2 * o, l, m, i, x); else update(2 * o + 1, m + 1, r, i, x);
11     mx[o] = max(mx[2 * o], mx[2 * o + 1]);
12 }
13 int firstAtLeast(int o, int l, int r, int L, int x) {   // first i >= L with a[i] >= x
14     if (r < L || mx[o] < x) return -1;
15     if (l == r) return l;
16     int m = (l + r) / 2;
17     int res = firstAtLeast(2 * o, l, m, L, x);
18     if (res != -1) return res;
19     return firstAtLeast(2 * o + 1, m + 1, r, L, x);
20 }
21  
22 int main() {
23     int q;
24     cin >> n >> q;
25     for (int i = 1; i <= n; i++) { int x; cin >> x; update(1, 1, n, i, x); }
26     for (int k = 0; k < q; k++) {
27         int op, a, b;
28         cin >> op >> a >> b;
29         if (op == 1) update(1, 1, n, a, b);
30         else cout << firstAtLeast(1, 1, n, a, b) << "\n";
31     }

Key	points	to	memorise
1. Fenwick walk: for step from high to low: if t[p+step] < k then p += step, k −= t[p]; answer p + 1.
2. DFS order: subtree = [tin, tout]; range-add/range-sum with two Fenwick trees.
3. Root path sums: +v at tin, −v at tout + 1, point query at tin.
4. Persistent trie: copy one path per insert; count(r) − count(l − 1) per node.
5. Prefix basis: at each bit keep the vector with the larger position; query uses pos ≥ l.
6. Segment tree descent: skip nodes out of range or failing; left first; O(log n).
