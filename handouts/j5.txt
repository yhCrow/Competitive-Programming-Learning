IDEA A binary	heap ( 二叉堆 ) is a complete
binary tree stored in an array (children
of i at 2i and 2i + 1, 基础篇  第 16 章 )
with one rule: every parent ≤ its
children. The minimum is always at
h[1]. Insert and delete-min each walk
one root-to-leaf path: O(log n).
KEY POINT Insert: put the value at the end, then sift
up (swap with the parent while
smaller). Delete-min: move the last
value to the root, then sift	down (swap
with the smaller child while larger).
深 入 浅 出 程 序 设 计 竞 赛  · 进 阶 篇  · 第 2 部 分  进 阶 数 据 结 构
Heaps for the smallest (or largest) element in O(log n), and Fenwick trees for prefix sums that can change.

heap.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 int h[1000005], sz = 0;
5  
6 void push(int x) {
7     h[++sz] = x;
8     for (int i = sz; i > 1 && h[i] < h[i / 2]; i /= 2) swap(h[i], h[i / 2]);   // sift up
9 }
10  
11 void pop() {
12     h[1] = h[sz--];                                  // last value to the root
13     int i = 1;
14     while (2 * i <= sz) {                            // sift down
15         int c = 2 * i;
16         if (c + 1 <= sz && h[c + 1] < h[c]) c++;     // the smaller child
17         if (h[i] <= h[c]) break;
18         swap(h[i], h[c]);
19         i = c;
20     }
21 }
22  
23 int main() {
24     ios::sync_with_stdio(false);
25     cin.tie(nullptr);
26     int n;
27     cin >> n;
28     while (n--) {
29         int op;
30         cin >> op;
31         if (op == 1) { int x; cin >> x; push(x); }
32         else if (op == 2) cout << h[1] << "\n";      // the minimum
33         else pop();
34     }
35     return 0;
36 }

KEY POINT The tree has height log₂ n, so sift up and sift down each take O(log n) swaps; reading the minimum is O(1)
(P3378). Building a heap from n values by sifting down from n/2 to 1 is even O(n).
5.2 优先队列
priority_queue:	two-heap	median,	k	smallest	sums,	deadlines.
IDEA C++ already has a heap: priority_queue<T>  keeps the largest on top; priority_queue<T, vector<T>,
greater<T>>  keeps the smallest. push , pop  O(log n), top  O(1). Pairs compare by the first field, then the
second.

running_median.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 int main() {
5     int n;
6     cin >> n;
7     priority_queue<int> low;                              // smaller half, max on top
8     priority_queue<int, vector<int>, greater<int>> high;  // larger half, min on top
9     for (int i = 1; i <= n; i++) {
10         int x;
11         cin >> x;
12         if (low.empty() || x <= low.top()) low.push(x);
13         else high.push(x);
14         if (low.size() > high.size() + 1) { high.push(low.top()); low.pop(); }  // rebalance
15         if (high.size() > low.size()) { low.push(high.top()); high.pop(); }
16         if (i % 2 == 1) cout << low.top() << "\n";       // median of the first i
17     }
18     return 0;
19 }

IDEA Keep the smaller half in a max-heap and the larger half in a min-heap, with sizes equal or the small half
one bigger. The median is then the top of the small half (P1168). The same two-heap split answers "the k-
th smallest so far" with a moving k (P1801).

repairs.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 int main() {
5     int n;
6     cin >> n;
7     vector<pair<long long, long long>> b(n);       // (deadline, time needed)
8     for (int i = 0; i < n; i++) cin >> b[i].second >> b[i].first;
9     sort(b.begin(), b.end());                      // earliest deadline first
10     priority_queue<long long> taken;               // times of repairs we do
11     long long now = 0;
12     for (int i = 0; i < n; i++) {
13         now += b[i].second;
14         taken.push(b[i].second);
15         if (now > b[i].first) {                    // too late: drop the longest repair
16             now -= taken.top();
17             taken.pop();
18         }
19     }
20     cout << taken.size() << "\n";
21     return 0;
22 }

KEY POINT The n smallest of all n² sums: row i is sorted, so only the next unused element of each row can be next; a
heap of n "row fronts" gives each answer in O(log n) (P1631, also P2085). Deadlines: take jobs by
deadline, and when time runs out drop the longest job taken so far, found in O(log n) with a max-heap
(P4053).
KEY POINT Huffman with k-ary trees (P2168) adds zero-weight leaves so that every merge takes exactly k;
earthworms (P2827) replace the heap by three queues that stay sorted on their own; dancing pairs
(P1878) keep candidate adjacent pairs in a heap and skip stale ones.

IDEA A Fenwick	tree ( 树状数组 , BIT) stores
t[i] = sum of a over the block (i −
lowbit(i), i], where lowbit(i)	=	i	&	−i is
the lowest set bit. A prefix sum adds
about log n blocks, jumping i −=
lowbit(i); a point update changes about
log n blocks, jumping i += lowbit(i).
KEY POINT Prefix sums ( 进阶篇  第 2 章 ) answer
queries in O(1) but cannot change; a
Fenwick tree does both query and
update in O(log n), with a 10-line
implementation.

1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 int n;
5 long long t[500005];
6  
7 void add(int i, long long v) { for (; i <= n; i += i & -i) t[i] += v; }
8 long long sum(int i) { long long s = 0; for (; i > 0; i -= i & -i) s += t[i]; return s; }
9  
10 int main() {
11     ios::sync_with_stdio(false);
12     cin.tie(nullptr);
13     int m;
14     cin >> n >> m;
15     long long prev = 0;
16     for (int i = 1; i <= n; i++) {
17         long long x;
18         cin >> x;
19         add(i, x - prev);                  // store the difference array
20         prev = x;
21     }
22     while (m--) {
23         int op;
24         cin >> op;
25         if (op == 1) {
26             int l, r; long long k;
27             cin >> l >> r >> k;
28             add(l, k); add(r + 1, -k);     // range add = two point changes
29         } else {
30             int x; cin >> x;
31             cout << sum(x) << "\n";        // prefix of differences = a[x]
32         }
33     }
34     return 0;
35 }

inversions_bit.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 int n;
5 long long t[500005];
6  
7 void add(int i) { for (; i <= n; i += i & -i) t[i]++; }
8 long long sum(int i) { long long s = 0; for (; i > 0; i -= i & -i) s += t[i]; return s; }
9  
10 int main() {
11     ios::sync_with_stdio(false);
12     cin.tie(nullptr);
13     cin >> n;
14     vector<int> a(n), v;
15     for (int i = 0; i < n; i++) cin >> a[i];
16     v = a;
17     sort(v.begin(), v.end());
18     v.erase(unique(v.begin(), v.end()), v.end());      // discretise to 1..k
19     long long inv = 0;
20     for (int i = 0; i < n; i++) {
21         int r = lower_bound(v.begin(), v.end(), a[i]) - v.begin() + 1;
22         inv += i - sum(r);                             // earlier values > a[i]
23         add(r);
24     }
25     cout << inv << "\n";
26     return 0;
27 }

INPUT 2

IDEA Inversions again (P1908, also 第 3 章  by merge sort): scan left to right, and before inserting a[i] ask how
many earlier values are ≤ a[i]; the rest of the i earlier values are larger. Discretise first so the values index
the tree.
KEY POINT Equal values are not inversions: count earlier values ≤ a[i] (not <), as above. Matching match-sticks
(P1966), counting close pairs (P5677) and booking a hall (P2161) all reduce to counting with a Fenwick
tree.

Key	points	to	memorise
1. Heap: parent ≤ children; min at h[1]; children 2i, 2i + 1.
2. Insert: append, sift up. Delete-min: last to root, sift down with the smaller child.
3. priority_queue: max on top; greater<T> for min; push/pop O(log n).
4. Median: max-heap of the small half + min-heap of the large half, balanced.
5. n smallest of many sorted rows: heap of row fronts.
6. Deadline scheduling: by deadline; when late, drop the longest taken job.
7. Fenwick: lowbit(i)	=	i	&	−i; update i += lowbit, query i −= lowbit; O(log n).
8. Range add + point query: Fenwick on the difference array.
9. Inversions: discretise; before inserting a[i], add i − (count ≤ a[i]).
