IDEA With m queries on n numbers, adding each range costs O(nm); prefix sums cost O(n + m). This is the first
tool to try whenever a problem asks for many range sums (P8218).
2.1.1 Two-dimensional	prefix	sums
S[i][j] = sum of the rectangle from (1, 1) to (i, j). Build and query by inclusion–exclusion:
S[i][j] = a[i][j] + S[i-1][j] + S[i][j-1] - S[i-1][j-1];
 
// sum of rows x1..x2, columns y1..y2
S[x2][y2] - S[x1-1][y2] - S[x2][y1-1] + S[x1-1][y1-1]
KEY POINT The corner S[i−1][j−1] is counted twice when adding the strip above and the strip to the left, so subtract it
once. In a query, the corner is subtracted twice, so add it back.

max_rectangle.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 long long col[125][125];                  // col[i][j]: column j, rows 1..i
5  
6 int main() {
7     int n;
8     cin >> n;
9     for (int i = 1; i <= n; i++)
10         for (int j = 1; j <= n; j++) {
11             long long x;
12             cin >> x;
13             col[i][j] = col[i - 1][j] + x;
14         }
15     long long best = LLONG_MIN;
16     for (int top = 1; top <= n; top++)
17         for (int bot = top; bot <= n; bot++) {   // fix the rows,
18             long long here = 0;                   // then max subarray on columns
19             for (int j = 1; j <= n; j++) {
20                 long long v = col[bot][j] - col[top - 1][j];
21                 here = max(here + v, v);
22                 best = max(best, here);
23             }
24         }
25     cout << best << "\n";
26     return 0;
27 }

IDEA Maximum-sum rectangle: fix the top and bottom rows (O(n²) pairs); the column sums between them form
a 1D array, and the best range of columns is the maximum subarray of 第 1 章 . Column prefix sums give
each value in O(1): O(n³) in total (P1719).
KEY POINT Binary search on a parameter, with prefix sums so that each check is O(n), solves the quality inspector
(P1314); splitting brownies into an even grid checks each guess with 2D prefix sums (P3017).

IDEA Difference	array ( 差分 ): d[i] = a[i] −
a[i − 1]. Adding v to a whole range a[l..r]
changes only two entries: d[l]	+=	v,	d[r
+	1]	−=	v. After all updates, a prefix sum
over d rebuilds a. Range updates
become O(1) each.
KEY POINT Prefix sums and differences undo each
other: prefix sums answer many range
queries, differences apply many range
updates. Use differences when all
updates come before all queries.

1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 long long a[5000005], dif[5000005];
5  
6 int main() {
7     ios::sync_with_stdio(false);
8     cin.tie(nullptr);
9     int n, p;
10     cin >> n >> p;
11     for (int i = 1; i <= n; i++) cin >> a[i];
12     for (int i = 1; i <= n; i++) dif[i] = a[i] - a[i - 1];
13     while (p--) {
14         int x, y;
15         long long z;
16         cin >> x >> y >> z;
17         dif[x] += z;                 // range [x, y] gets +z
18         dif[y + 1] -= z;
19     }
20     long long cur = 0, low = LLONG_MAX;
21     for (int i = 1; i <= n; i++) {
22         cur += dif[i];               // rebuild a[i]
23         low = min(low, cur);
24     }

1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 int main() {
5     int n;
6     cin >> n;
7     vector<long long> a(n + 1);
8     for (int i = 1; i <= n; i++) cin >> a[i];
9     long long pos = 0, neg = 0;
10     for (int i = 2; i <= n; i++) {
11         long long d = a[i] - a[i - 1];
12         if (d > 0) pos += d;
13         else neg -= d;
14     }
15     cout << max(pos, neg) << "\n";        // fewest operations
16     cout << abs(pos - neg) + 1 << "\n";   // possible final values
17     return 0;
18 }

KEY POINT Changing a range by ±1 changes one difference up and one down (or one at the ends). To make all values
equal, every d[2..n] must become 0: pair a positive with a negative while both exist, then fix the rest alone,
so the answer is max(pos, neg) operations (P4552). Ticket counting per rail segment (P3406) and
checking orders for classrooms with binary search (P1083) are difference arrays too.

2.2.1 Two-dimensional	differences
carpets.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 int dif[1005][1005];
5  
6 int main() {
7     int n, m;
8     cin >> n >> m;
9     while (m--) {
10         int x1, y1, x2, y2;
11         cin >> x1 >> y1 >> x2 >> y2;
12         dif[x1][y1]++;                    // +1 on the rectangle:
13         dif[x1][y2 + 1]--;                // four corners
14         dif[x2 + 1][y1]--;
15         dif[x2 + 1][y2 + 1]++;
16     }
17     for (int i = 1; i <= n; i++) {
18         for (int j = 1; j <= n; j++) {
19             dif[i][j] += dif[i - 1][j] + dif[i][j - 1] - dif[i - 1][j - 1];
20             cout << dif[i][j] << " ";     // 2D prefix sum rebuilds counts
21         }
22         cout << "\n";
23     }
24     return 0;
25 }

IDEA Adding 1 to a rectangle touches four corners of the 2D difference array. A 2D prefix sum afterwards turns
the corners back into the full counts: O(1) per carpet and O(n²) once at the end, instead of O(n²) per carpet
(P3397).
KEY POINT Flipping k cows at a time with as few flips as possible keeps a running difference of "flips still active"
while scanning (P2882).

IDEA Discretisation ( 离散化 ): when only the
order of values matters, not their size,
replace each value by its rank among the
distinct values. Coordinates up to 109
become 0 … n − 1, small enough to index
an array.
KEY POINT Recipe: copy the values, sort , unique
+ erase , then rank(x) =
lower_bound(v.begin(), v.end(),
x) - v.begin() . O(n log n).
burning_ships.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 int main() {
5     int n;
6     cin >> n;
7     vector<long long> L(n), R(n), xs;
8     for (int i = 0; i < n; i++) {
9         cin >> L[i] >> R[i];                  // burning interval [L, R)
10         xs.push_back(L[i]);
11         xs.push_back(R[i]);
12     }
13     sort(xs.begin(), xs.end());
14     xs.erase(unique(xs.begin(), xs.end()), xs.end());
15     int k = xs.size();
16     vector<int> dif(k + 1, 0);
17     for (int i = 0; i < n; i++) {
18         int a = lower_bound(xs.begin(), xs.end(), L[i]) - xs.begin();
19         int b = lower_bound(xs.begin(), xs.end(), R[i]) - xs.begin();
20         dif[a]++;                             // pieces a .. b-1 are covered
21         dif[b]--;
22     }
23     long long total = 0;
24     int cover = 0;
25     for (int i = 0; i + 1 < k; i++) {
26         cover += dif[i];
27         if (cover > 0) total += xs[i + 1] - xs[i];   // real length of piece i
28     }
29     cout << total << "\n";
30     return 0;
31 }

IDEA The sorted endpoints cut the line into at most 2n pieces; inside one piece nothing changes. A difference
array over the piece	numbers counts how many intervals cover each piece, and covered pieces add their
real length xs[i + 1] − xs[i] (P1496).

overplanting.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 int cover[2005][2005];
5  
6 int main() {
7     int n;
8     cin >> n;
9     vector<long long> X1(n), Y1(n), X2(n), Y2(n), xs, ys;
10     for (int i = 0; i < n; i++) {
11         cin >> X1[i] >> Y1[i] >> X2[i] >> Y2[i];   // top-left, bottom-right
12         swap(Y1[i], Y2[i]);                          // now Y1 < Y2
13         xs.push_back(X1[i]); xs.push_back(X2[i]);
14         ys.push_back(Y1[i]); ys.push_back(Y2[i]);
15     }
16     sort(xs.begin(), xs.end()); xs.erase(unique(xs.begin(), xs.end()), xs.end());
17     sort(ys.begin(), ys.end()); ys.erase(unique(ys.begin(), ys.end()), ys.end());
18     for (int i = 0; i < n; i++) {                    // 2D difference on ranks
19         int a = lower_bound(xs.begin(), xs.end(), X1[i]) - xs.begin() + 1;
20         int b = lower_bound(xs.begin(), xs.end(), X2[i]) - xs.begin() + 1;
21         int c = lower_bound(ys.begin(), ys.end(), Y1[i]) - ys.begin() + 1;
22         int e = lower_bound(ys.begin(), ys.end(), Y2[i]) - ys.begin() + 1;
23         cover[a][c]++; cover[a][e]--; cover[b][c]--; cover[b][e]++;
24     }
25     long long area = 0;
26     for (int i = 1; i < int(xs.size()); i++)
27         for (int j = 1; j < int(ys.size()); j++) {
28             cover[i][j] += cover[i - 1][j] + cover[i][j - 1] - cover[i - 1][j - 1];
29             if (cover[i][j] > 0)
30                 area += (xs[i] - xs[i - 1]) * (ys[j] - ys[j - 1]);   // real cell size
31         }
32     cout << area << "\n";
33     return 0;
34 }
INPUT
2
0 5 4 1
2 4 6 2
OUTPUT
20
KEY POINT In 2D, the sorted x and y coordinates cut the plane into at most 2n × 2n cells; a 2D difference marks which
cells are covered and each covered cell adds its real area (P1884). Program analysis with huge variable
numbers discretises before union-find (P1955); skylines (P1904), cow line-ups (P3029) and parity games
(P5937) compress coordinates the same way.
TRAP After discretising, distance is lost: ranks 3 and 4 may be far apart in reality. Always add back the real
length (xs[i + 1] − xs[i] ) when measuring, and keep both ends of every interval in the list.

Key	points	to	memorise
1. Prefix sums: S[i]	=	S[i−1]	+	a[i]; range sum S[r]	−	S[l−1]; 1-based, long long.
2. 2D: S[i][j] = a + S[i−1][j] + S[i][j−1] − S[i−1][j−1].
3. 2D query: S[x2][y2] − S[x1−1][y2] − S[x2][y1−1] + S[x1−1][y1−1].
4. Max rectangle: fix two rows, max subarray on the column sums, O(n³).
5. Difference: add v to [l, r] → d[l]	+=	v,	d[r+1]	−=	v; prefix sum rebuilds a.
6. 2D difference: +1 at (x1,y1), (x2+1,y2+1); −1 at (x1,y2+1), (x2+1,y1).
7. Equalise with ±1 range moves: max(Σ positive diffs, Σ negative diffs).
8. Discretise: sort, unique, erase; rank = lower_bound − begin.
9. After discretising, add back real lengths xs[i+1] − xs[i].
