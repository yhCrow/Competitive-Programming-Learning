18.1 扫描线
Events	+	covered	length	per	node.
IDEA The sweep	line ( 扫描线 ) turns a 2D problem into many 1D ones. For the area of a union of rectangles,
sweep a vertical line from left to right. Each rectangle becomes two events: at x1 add +1 on its y-interval, at
x2 add −1. Between two consecutive events the covered length on the line stays constant, so area +=
covered length × Δx (P5490).
KEY POINT A segment tree over the (discretised) y-coordinates stores, per node, cnt = how many whole rectangles
cover the node's interval, and len = covered length inside it. No lazy push is needed: if cnt > 0 the whole
interval is covered, else len = sum of the children. Every +1 has a matching −1, which is why this works.
竞 赛 拓 展 专 题  · 拓 展 篇  · 数 据 结 构 拓 展
Four more segment-tree patterns: sweep a line over rectangles, keep the best line at each midpoint, merge
per-node value trees up a tree, and build the Cartesian tree with a stack.

1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 const int N = 400005;
5 vector<long long> ys;
6 int cnt[N * 4];
7 long long len_[N * 4];
8 struct Ev { long long x, y1, y2; int d; };
9 bool byX(const Ev &a, const Ev &b) { return a.x < b.x; }
10  
11 void pull(int p, int l, int r) {                 // node covers ys[l] .. ys[r+1]
12     if (cnt[p] > 0) len_[p] = ys[r + 1] - ys[l];
13     else if (l == r) len_[p] = 0;
14     else len_[p] = len_[2 * p] + len_[2 * p + 1];
15 }
16 void update(int p, int l, int r, int x, int y, int d) {
17     if (y < l || r < x) return;
18     if (x <= l && r <= y) { cnt[p] += d; pull(p, l, r); return; }
19     int mid = (l + r) / 2;
20     update(2 * p, l, mid, x, y, d); update(2 * p + 1, mid + 1, r, x, y, d);
21     pull(p, l, r);
22 }

25     int n;

27     vector<Ev> ev;
28     for (int i = 0; i < n; i++) {
29         long long x1, y1, x2, y2;
30         cin >> x1 >> y1 >> x2 >> y2;
31         ev.push_back(Ev{x1, y1, y2, 1}); ev.push_back(Ev{x2, y1, y2, -1});
32         ys.push_back(y1); ys.push_back(y2);
33     }
34     sort(ys.begin(), ys.end()); ys.erase(unique(ys.begin(), ys.end()), ys.end());
35     sort(ev.begin(), ev.end(), byX);
36     int m = ys.size() - 1;                           // elementary y-intervals 0 .. m-1
37     long long area = 0;
38     for (int i = 0; i < int(ev.size()); i++) {
39         if (i > 0) area += len_[1] * (ev[i].x - ev[i - 1].x);   // strip since the last event
40         int a = lower_bound(ys.begin(), ys.end(), ev[i].y1) - ys.begin();
41         int b = lower_bound(ys.begin(), ys.end(), ev[i].y2) - ys.begin() - 1;
42         update(1, 0, m - 1, a, b, ev[i].d);
43     }
44     cout << area << "\n";
45     return 0;
46 }

18.2 李超线段树
Best	line	at	the	midpoint;	loser	goes	one	way.
IDEA A Li	Chao	tree ( 李超线段树 ) stores lines y = kx + b and answers "the maximum (or minimum) value at x"
in O(log C). Each node keeps the line that is best at its midpoint. When a new line arrives, compare at the
midpoint and swap so the better one stays; the loser can only win on one side, so push it into that child
only.
KEY POINT That "one side only" fact makes insertion O(log C) instead of touching both halves. It is the general
replacement for the convex hull trick ( 进阶篇  第 18 章 ): lines may arrive in any order and queries can be
anywhere. Segments (P4097) are inserted into O(log C) nodes first.

1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 const int LO = -100000, HI = 100000;
5 struct Line { long long k, b; long long at(long long x) const { return k * x + b; } };
6 vector<Line> t;                                  // node line
7 vector<int> lc, rc;
8 vector<bool> has;
9  
10 int newNode() { t.push_back(Line{0, 0}); lc.push_back(-1); rc.push_back(-1); has.push_back(false); 
return t.size() - 1; }
11  
12 void insert(int p, int l, int r, Line ln) {
13     if (!has[p]) { t[p] = ln; has[p] = true; return; }
14     int mid = l + (r - l) / 2;                   // floor for negatives too
15     if (ln.at(mid) > t[p].at(mid)) swap(ln, t[p]);   // keep the better one at mid
16     if (l == r) return;
17     if (ln.at(l) > t[p].at(l)) {                 // the loser can only win on the left
18         if (lc[p] < 0) { int c = newNode(); lc[p] = c; }
19         insert(lc[p], l, mid, ln);
20     } else if (ln.at(r) > t[p].at(r)) {          // ... or only on the right
21         if (rc[p] < 0) { int c = newNode(); rc[p] = c; }
22         insert(rc[p], mid + 1, r, ln);
23     }
24 }
25 long long query(int p, int l, int r, long long x) {
26     if (p < 0 || !has[p]) return LLONG_MIN;
27     long long res = t[p].at(x);
28     if (l == r) return res;
29     int mid = l + (r - l) / 2;
30     return max(res, x <= mid ? query(lc[p], l, mid, x) : query(rc[p], mid + 1, r, x));
31 }

38     while (q--) { long long x; cin >> x; cout << query(0, LO, HI, x) << "\n"; }
39     return 0;
40 }

18.3 线段树合并
Dynamic	value	trees	merged	up	the	tree.
IDEA Segment	tree	merging ( 线段树合并 ): give every tree node its own dynamic value segment tree (created
on demand). To combine children into a parent, merge their trees: walk both at once; where only one has a
node, take it as it is; where both have a node, merge their children recursively and pull up.
KEY POINT Each merge step destroys one node, so all merges together cost the total number of nodes created: O(n log
V). It answers "for every subtree" questions about value distributions (most frequent value, k-th
smallest), and with tree differences, path updates ( 雨天的尾巴  P4556).

seg_merge.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 const int N = 100005, V = 100000, M = N * 20;
5 int ls[M], rs[M], mx[M], who[M], tot = 0;     // mx: highest count in range, who: smallest value with it
6 int root[N], val[N], ans[N];
7 vector<int> g[N];
8  
9 void pull(int p) {
10     if (mx[ls[p]] >= mx[rs[p]]) { mx[p] = mx[ls[p]]; who[p] = who[ls[p]]; }   // tie: smaller value
11     else { mx[p] = mx[rs[p]]; who[p] = who[rs[p]]; }
12 }
13 int add(int p, int l, int r, int x) {          // count of value x += 1
14     if (!p) p = ++tot;
15     if (l == r) { mx[p]++; who[p] = l; return p; }
16     int mid = (l + r) / 2;
17     if (x <= mid) ls[p] = add(ls[p], l, mid, x); else rs[p] = add(rs[p], mid + 1, r, x);
18     pull(p); return p;
19 }
20 int merge(int a, int b, int l, int r) {
21     if (!a || !b) return a | b;                  // only one side: reuse it
22     if (l == r) { mx[a] += mx[b]; return a; }
23     int mid = (l + r) / 2;
24     ls[a] = merge(ls[a], ls[b], l, mid);
25     rs[a] = merge(rs[a], rs[b], mid + 1, r);
26     pull(a); return a;
27 }
28 void dfs(int u, int p) {
29     root[u] = add(0, 1, V, val[u]);
30     for (int k = 0; k < int(g[u].size()); k++) {
31         int v = g[u][k];
32         if (v == p) continue;
33         dfs(v, u);
34         root[u] = merge(root[u], root[v], 1, V);
35     }
36     ans[u] = who[root[u]];
37 }
38  
39 int main() {
40     int n;
41     cin >> n;
42     for (int i = 1; i <= n; i++) cin >> val[i];
43     for (int i = 1; i < n; i++) { int a, b; cin >> a >> b; g[a].push_back(b); g[b].push_back(a); }
44     dfs(1, 0);
45     for (int i = 1; i <= n; i++) cout << ans[i] << " ";
46     cout << "\n";
47     return 0;
48 }

18.4 笛卡尔树
Heap	by	value,	BST	by	index,	stack	build.
IDEA A Cartesian	tree ( 笛卡尔树 ) of an array: the root is the position of the minimum; the left subtree is built
from the part to its left, the right from the part to its right. It is a heap by value and a BST by index. Build it
in O(n) with a stack holding the rightmost chain.
KEY POINT Range minimum of [l, r] = the LCA of l and r in this tree. "For each element, the maximal range where it is
the minimum" is exactly its subtree, which is the same information a monotonic stack gives (largest
rectangle in a histogram, counting subarrays by their minimum).
cartesian.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 int main() {
5     int n;
6     cin >> n;
7     vector<int> a(n + 1), lc(n + 1, 0), rc(n + 1, 0), par(n + 1, 0), st;
8     for (int i = 1; i <= n; i++) cin >> a[i];
9     for (int i = 1; i <= n; i++) {
10         int last = 0;
11         while (!st.empty() && a[st.back()] > a[i]) { last = st.back(); st.pop_back(); }   // larger ones 
go below i
12         lc[i] = last;
13         if (!st.empty()) rc[st.back()] = i;
14         st.push_back(i);
15     }
16     for (int i = 1; i <= n; i++) {
17         if (lc[i]) par[lc[i]] = i;
18         if (rc[i]) par[rc[i]] = i;
19     }
20     for (int i = 1; i <= n; i++) cout << par[i] << " ";        // 0 = root
21     cout << "\n";
22     return 0;
23 }
INPUT
8
5 2 8 1 6 3 9 4
OUTPUT
2 4 2 0 6 4 8 6
IDEA Parents (1-based, 0 for the root) checked against a direct recursive build in Python. Every element is
pushed and popped at most once: O(n).
KEY POINT The popped chain becomes the new node's left child, and the new node becomes the right child of
whatever is left on the stack. With equal values decide a tie rule (strict > above) and keep it consistent.

Key	points	to	memorise
1. Sweep: +1 / −1 events; area += covered length × Δx.
2. Covered length: cnt > 0 ⇒ full interval, else sum of children; no lazy push.
3. Li Chao: keep the line best at mid; the loser can win on one side only.
4. Merging: one side empty ⇒ reuse; both ⇒ merge children; O(total nodes).
5. Read a node's answer right after its merges.
6. Cartesian tree: stack of the right chain; popped chain becomes the left child.
7. RMQ(l, r) = LCA in the Cartesian tree.
