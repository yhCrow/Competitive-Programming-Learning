5.1 分块
Partial	ends	+	whole	blocks;	lazy	adds;	sorted	blocks.

query [2, 10]: ends one by one (blue),

IDEA Sqrt	decomposition ( 分块 ): cut the
array into blocks of size about √n. A
range query or update touches at most
two partial blocks, handled element by
element, and the whole blocks in
between, handled by one value per
block (a sum, a lazy add, a sorted copy).
Each operation costs O(√n).
KEY POINT Segment trees are faster, but blocks are
far more flexible: anything you can keep
per block (sorted order, counts, a hash
table) works, and the code is a few
simple loops. When the information
does not merge nicely, think of blocks.
竞 赛 拓 展 专 题  · 拓 展 篇  · 数 据 结 构 拓 展
Cut the array into √n blocks to answer queries a segment tree cannot merge, and answer many offline
range queries by moving a window in a clever order.

block_sum.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 int n, m, B, bl[100005];
5 long long a[100005], sum[400], add_[400];
6  
7 void update(int l, int r, long long v) {
8     if (bl[l] == bl[r]) { for (int i = l; i <= r; i++) { a[i] += v; sum[bl[i]] += v; } return; }
9     for (int i = l; bl[i] == bl[l]; i++) { a[i] += v; sum[bl[i]] += v; }        // left end
10     for (int i = r; bl[i] == bl[r]; i--) { a[i] += v; sum[bl[i]] += v; }        // right end
11     for (int b = bl[l] + 1; b < bl[r]; b++) add_[b] += v;                       // whole blocks
12 }
13 long long query(int l, int r) {
14     long long s = 0;
15     if (bl[l] == bl[r]) { for (int i = l; i <= r; i++) s += a[i] + add_[bl[i]]; return s; }
16     for (int i = l; bl[i] == bl[l]; i++) s += a[i] + add_[bl[i]];
17     for (int i = r; bl[i] == bl[r]; i--) s += a[i] + add_[bl[i]];
18     for (int b = bl[l] + 1; b < bl[r]; b++) s += sum[b] + add_[b] * B;          // full block of size B
19     return s;
20 }
21  
22 int main() {
23     cin >> n >> m;
24     B = max(1, int(sqrt(n)));
25     for (int i = 1; i <= n; i++) { cin >> a[i]; bl[i] = (i - 1) / B; sum[bl[i]] += a[i]; }
26     while (m--) {
27         int op, x, y;
28         cin >> op >> x >> y;
29         if (op == 1) { long long k; cin >> k; update(x, y, k); }
30         else cout << query(x, y) << "\n";
31     }
32     return 0;
33 }

IDEA Range add + range sum (P3372, the segment tree template) with blocks. bl[i] is the block of position i.
Whole blocks only get a lazy add_; partial blocks update elements and the block sum directly.
KEY POINT Whole blocks lie strictly between bl[l] and bl[r], so they always have the full size B; the last, shorter
block is only ever a partial block.

magic.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 int n, q, B, bl[1000005];
5 long long a[1000005], add_[1005];
6 vector<long long> sorted_[1005];                 // each block, sorted (without its add_)
7  
8 void rebuild(int b) {
9     sorted_[b].clear();
10     for (int i = b * B + 1; i <= min(n, (b + 1) * B); i++) sorted_[b].push_back(a[i]);
11     sort(sorted_[b].begin(), sorted_[b].end());
12 }
13  
14 int main() {
15     cin >> n >> q;
16     B = max(1, int(sqrt(n)));
17     for (int i = 1; i <= n; i++) { cin >> a[i]; bl[i] = (i - 1) / B; }
18     for (int b = 0; b <= bl[n]; b++) rebuild(b);
19     while (q--) {
20         char op; int l, r; long long w;
21         cin >> op >> l >> r >> w;
22         if (op == 'M') {                         // add w to [l, r]
23             if (bl[l] == bl[r]) { for (int i = l; i <= r; i++) a[i] += w; rebuild(bl[l]); continue; }
24             for (int i = l; bl[i] == bl[l]; i++) a[i] += w;
25             for (int i = r; bl[i] == bl[r]; i--) a[i] += w;
26             rebuild(bl[l]); rebuild(bl[r]);
27             for (int b = bl[l] + 1; b < bl[r]; b++) add_[b] += w;    // order inside is unchanged
28         } else {                                 // how many in [l, r] are >= w
29             int cnt = 0;
30             if (bl[l] == bl[r]) { for (int i = l; i <= r; i++) cnt += a[i] + add_[bl[i]] >= w; }
31             else {
32                 for (int i = l; bl[i] == bl[l]; i++) cnt += a[i] + add_[bl[i]] >= w;
33                 for (int i = r; bl[i] == bl[r]; i--) cnt += a[i] + add_[bl[i]] >= w;
34                 for (int b = bl[l] + 1; b < bl[r]; b++)          // binary search in the sorted block
35                     cnt += sorted_[b].end() - lower_bound(sorted_[b].begin(), sorted_[b].end(), w - 
add_[b]);
36             }
37             cout << cnt << "\n";
38         }
39     }
40     return 0;
41 }

IDEA The magician (P2801): range add, and "how many in [l, r] are at least w". A segment tree cannot merge
such counts, but a sorted	copy of each block answers it by binary search. Adding to a whole block keeps
its order, so only the two partial blocks are re-sorted.
KEY POINT Query cost √n · log n, update cost √n log √n. Choosing the block size to balance the two parts is part of
the design; √n is the usual start.

5.2 莫队算法
Sorted	offline	queries;	moving	window;	variants.
IDEA Mo's	algorithm ( 莫队 ) answers many offline range queries when moving one end by one position is
cheap (add or remove one element). Sort the queries by (block of l, then r), and move a window [L, R]
from one query to the next, adjusting one element at a time.
KEY POINT Inside one block of l, r only increases: O(n) per block, √n blocks. l moves at most √n per query. Total
O((n + q)√n). It only needs add(x) and remove(x) in O(1): counts of values, number of distinct values,
sums of squares, pairs of equal values.
mo.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 int n, m, k, B, a[50005], cnt[50005];
5 long long cur = 0, ans[50005];
6 struct Q { int l, r, id; };
7  
8 bool cmp(const Q &x, const Q &y) {
9     if (x.l / B != y.l / B) return x.l / B < y.l / B;
10     return (x.l / B) % 2 ? x.r > y.r : x.r < y.r;     // odd blocks: r goes back (fewer moves)
11 }
12 void add(int i) { cur += 2 * cnt[a[i]] + 1; cnt[a[i]]++; }      // (c+1)^2 - c^2
13 void remove_(int i) { cnt[a[i]]--; cur -= 2 * cnt[a[i]] + 1; }
14  
15 int main() {
16     cin >> n >> m >> k;
17     B = max(1, int(sqrt(n)));
18     for (int i = 1; i <= n; i++) cin >> a[i];
19     vector<Q> qs(m);
20     for (int i = 0; i < m; i++) { cin >> qs[i].l >> qs[i].r; qs[i].id = i; }
21     sort(qs.begin(), qs.end(), cmp);
22     int L = 1, R = 0;                                 // empty window
23     for (int i = 0; i < m; i++) {
24         while (R < qs[i].r) add(++R);                 // grow first,
25         while (L > qs[i].l) add(--L);
26         while (R > qs[i].r) remove_(R--);             // then shrink
27         while (L < qs[i].l) remove_(L++);
28         ans[qs[i].id] = cur;
29     }
30     for (int i = 0; i < m; i++) cout << ans[i] << "\n";
31     return 0;
32 }
INPUT
6 4 3
1 3 2 1 1 3
1 4
2 6
3 5
5 6
OUTPUT
6
9
5
2
IDEA Small B's question (P2709): Σ (count of value v in [l, r])². Adding one copy of a value with count c changes
the sum by (c + 1)² − c² = 2c + 1; removing reverses it.

5.3 课后习题与实验
The	Luogu	problems	for	this	chapter,	a	ladder	through	them,	and	experiments	to	try	yourself.
Key	points	to	memorise
1. Blocks of about √n: two partial ends element by element, whole blocks by stored values.
2. Whole blocks can keep sums, lazy adds, sorted copies, counts …
3. Adding to a whole block keeps its sorted order; re-sort only partial blocks.
4. Mo: sort by (block of l, r); move L and R one step at a time; O((n + q)√n).
5. Grow before shrink; store answers by original index.
6. Updates: time as a 3rd coordinate, block n^(2/3). Hard remove: rollback Mo.
7. Mo is offline; forced-online problems need precomputed block tables.
