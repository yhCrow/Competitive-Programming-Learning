7.1 CDQ	 分治
Three-dimensional	partial	order;	time	as	a	coordinate.
left [l, mid] right [mid+1, r]
here: only left → right effects
IDEA CDQ	divide	and	conquer (CDQ 分治 )
handles problems where each element
is affected by "earlier" ones in one
coordinate. Split by that coordinate into
halves; solve each half recursively; then
count only the effects of the left half on
the right half. That middle step has one
coordinate fewer to worry about.
KEY POINT Three-dimensional partial order ( 三维
偏序 , P3810): count j with aj ≤ ai, bj ≤ bi,
cj ≤ ci. Sort by a (one dimension gone);
CDQ on positions (left half has smaller
a); in the middle step sort both halves
by b and sweep with a Fenwick tree on
c. O(n log² n).
竞 赛 拓 展 专 题  · 拓 展 篇  · 数 据 结 构 拓 展
Two offline tricks that replace heavy data structures: CDQ lets the left half act on the right half to peel off
one dimension, and overall binary search finds the answers of all queries together.

cdq.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 const int N = 100005, K = 200005;
5 struct E { int a, b, c, cnt, res; };
6 E e[N], tmp[N];
7 int n, k, m, bit[K], ans[N];
8  
9 void add(int i, int v) { for (; i <= k; i += i & -i) bit[i] += v; }
10 int sum(int i) { int s = 0; for (; i > 0; i -= i & -i) s += bit[i]; return s; }
11 bool byABC(const E &x, const E &y) {
12     if (x.a != y.a) return x.a < y.a;
13     if (x.b != y.b) return x.b < y.b;
14     return x.c < y.c;
15 }
16  
17 void cdq(int l, int r) {
18     if (l == r) return;
19     int mid = (l + r) / 2;
20     cdq(l, mid); cdq(mid + 1, r);                 // each half is now sorted by b
21     int i = l, p = l;
22     for (int j = mid + 1; j <= r; j++) {          // right element j collects every left with b <= b_j
23         while (i <= mid && e[i].b <= e[j].b) { add(e[i].c, e[i].cnt); tmp[p++] = e[i++]; }
24         e[j].res += sum(e[j].c);
25         tmp[p++] = e[j];
26     }
27     for (int t = l; t < i; t++) add(e[t].c, -e[t].cnt);   // undo the Fenwick changes
28     while (i <= mid) tmp[p++] = e[i++];
29     for (int t = l; t <= r; t++) e[t] = tmp[t];   // merge by b (merge sort on the way back)
30 }
31  
32 int main() {
33     cin >> n >> k;
34     vector<E> raw(n);
35     for (int i = 0; i < n; i++) { cin >> raw[i].a >> raw[i].b >> raw[i].c; raw[i].cnt = 1; raw[i].res = 
0; }
36     sort(raw.begin(), raw.end(), byABC);
37     m = 0;                                        // merge identical triples (they all count each other)
38     for (int i = 0; i < n; i++) {
39         if (m > 0 && e[m].a == raw[i].a && e[m].b == raw[i].b && e[m].c == raw[i].c) e[m].cnt++;
40         else e[++m] = raw[i];
41     }
42     cdq(1, m);
43     for (int i = 1; i <= m; i++) ans[e[i].res + e[i].cnt - 1] += e[i].cnt;   // equal copies count too
44     for (int d = 0; d < n; d++) cout << ans[d] << "\n";
45     return 0;
46 }

KEY POINT Output: for each d, how many elements are dominated-or-equal by exactly d others. Identical triples must
be merged first: copies dominate each other, and splitting them across halves would count only one
direction.
TRAP Undo the Fenwick tree with the same elements you added (not by clearing it all, which is O(k) per call).
The merge-by-b inside cdq replaces a separate sort, keeping the total at O(n log² n).
KEY POINT CDQ also turns dynamic problems into static ones: treat time as the first coordinate, so earlier updates
affect later queries (Mokia P4390: point adds and rectangle sums; 天使玩偶  P4169: nearest point; 动态逆
序对  P3157).
7.2 整体二分
Binary	search	all	queries	at	once	with	a	Fenwick	tree.
IDEA Overall	binary	search ( 整体二分 ) binary-searches the answers of all	queries	at	once. solve(lo, hi,
queries): every query here has its answer in [lo, hi]. Take mid, apply all updates with value ≤ mid (for
example: mark positions in a Fenwick tree), and check each query: if enough values ≤ mid lie in its range,
its answer is ≤ mid, else subtract that count and send it right.
KEY POINT Each level of the recursion handles every query and every element once, with a Fenwick tree: O((n + q) log
n · log V). It replaces a persistent or nested data structure with a simple Fenwick tree, and also works for
queries mixed with updates (P2617, P3332).

parallel_bs.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 const int N = 200005;
5 int n, m, bit[N], ans[N];
6 struct Item { int type, l, r, k, id; };      // type 0: value a[l] = k at position l; type 1: query
7  
8 void add(int i, int v) { for (; i <= n; i += i & -i) bit[i] += v; }
9 int sum(int i) { int s = 0; for (; i > 0; i -= i & -i) s += bit[i]; return s; }
10  
11 void solve(int lo, int hi, vector<Item> &v) {
12     if (v.empty()) return;
13     if (lo == hi) { for (int i = 0; i < int(v.size()); i++) if (v[i].type) ans[v[i].id] = lo; return; }
14     int mid = lo + (hi - lo) / 2;
15     vector<Item> L, R;
16     for (int i = 0; i < int(v.size()); i++) {
17         Item it = v[i];
18         if (!it.type) {                           // a number: does it go left (<= mid)?
19             if (it.k <= mid) { add(it.l, 1); L.push_back(it); } else R.push_back(it);
20         } else {
21             int c = sum(it.r) - sum(it.l - 1);    // numbers <= mid inside [l, r]
22             if (it.k <= c) L.push_back(it);
23             else { it.k -= c; R.push_back(it); }
24         }
25     }
26     for (int i = 0; i < int(L.size()); i++) if (!L[i].type) add(L[i].l, -1);   // undo
27     solve(lo, mid, L);
28     solve(mid + 1, hi, R);
29 }
30  
31 int main() {
32     cin >> n >> m;
33     vector<Item> v;
34     for (int i = 1; i <= n; i++) { int x; cin >> x; v.push_back(Item{0, i, i, x, 0}); }
35     for (int i = 0; i < m; i++) { int l, r, k; cin >> l >> r >> k; v.push_back(Item{1, l, r, k, i}); }
36     solve(-1000000000, 1000000000, v);
37     for (int i = 0; i < m; i++) cout << ans[i] << "\n";
38     return 0;
39 }

7.3 课后习题与实验
The	Luogu	problems	for	this	chapter,	a	ladder	through	them,	and	experiments	to	try	yourself.
Key	points	to	memorise
1. CDQ: solve halves, then count left → right effects only.
2. 3D partial order: sort by a, CDQ merges by b, Fenwick on c; O(n log² n).
3. Merge identical elements first; undo Fenwick changes element by element.
4. Time as the first coordinate turns online updates into a static problem.
5. Overall binary search: solve(lo, hi, items); items ≤ mid go into the Fenwick tree.
6. Query: k ≤ count ⇒ left; else k −= count, right. Keep items in time order.
