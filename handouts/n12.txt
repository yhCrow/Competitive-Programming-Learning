12.1 猫树
Split	node	by	the	highest	differing	bit.
IDEA A cat	tree ( 猫树 ) answers static range queries in O(1) for any associative merge whose partial results can
be precomputed. Build a segment tree over a power-of-two length. At each node with midpoint mid, store
the answer for every suffix [i, mid] of the left half and every prefix [mid + 1, j] of the right half. A query [l,
r] with l ≠ r uses the node where l and r are first split, which is found from the highest differing bit of l and
r.
KEY POINT O(n log n) memory and preprocessing, O(1) per query, no LCA walk. For maximum subarray sum each
stored entry holds the best inside, the best touching the middle, and the sum. Knapsack-like merges ( 好吃
的题目  P6240) store a whole DP array per position, so queries cost the merge of two arrays.
竞 赛 拓 展 专 题  · NOI 篇  · NOI 专 题
O(1) static range queries with precomputed half-answers, historic maxima with a "best pending prefix" tag,
full code for dynamic centroid decomposition, and the idea behind second-offline Mo.

23     }
24     build(lv + 1, l, mid); build(lv + 1, mid + 1, r);
25 }

12.2 区间历史最值
Tags	(add,	max	prefix	of	adds).
IDEA Historic	maximum ( 区间历史最值 ): besides the current values ai, ask for max over time of ai. With range
add, each node keeps mx (current max) and hmx (historic max), and two lazy tags: add (total pending
addition) and hadd (the largest prefix of that pending sequence of additions). When pushing tags (add,
hadd) onto a child: child.hmx = max(child.hmx, child.mx + hadd), child.hadd = max(child.hadd, child.add
+ hadd), then child.mx += add, child.add += add.
KEY POINT The order matters: the historic values must be updated with the child's old current values before adding.
That "max prefix of pending additions" tag is the whole trick; CPU 监控  (P4314) adds assignment tags the
same way, and P6242 combines it with chmin (NOI 篇  第 5 章 ).

8 void apply(int p, long long ad, long long had) {     // pending (ad, had) arrives at node p
9     hmx[p] = max(hmx[p], mx[p] + had);               // old current + best prefix
10     hadd[p] = max(hadd[p], add_[p] + had);
11     mx[p] += ad; add_[p] += ad;
12 }
13 void push(int p) {
14     apply(2 * p, add_[p], hadd[p]); apply(2 * p + 1, add_[p], hadd[p]);
15     add_[p] = hadd[p] = 0;

12.3 动态点分治（代码）
Two	Fenwick	trees	per	centroid.
IDEA Dynamic	centroid	decomposition with code ( 拓展篇  第 27 章  gave the idea): query "sum of values
within distance k of x" with point updates ( 震波  P6329). For each centroid c keep two Fenwick trees
indexed by distance: Fc[d] = total value at distance d from c inside c's piece, and Gc[d] = total value inside
c's piece at distance d from c's centroid	parent.
KEY POINT Query(x, k): start with Fx up to k; then for each centroid ancestor p of the current c: add Fp up to k − dist(x,
p), and subtract Gc up to k − dist(x, p) (those vertices were already counted below). Update changes O(log
n) Fenwick entries the same way. dist uses the original tree (LCA).

11 long long sumF(vector<long long> &t, int i) { if (i < 0) return 0; long long s = 0; for (i = min(i + 1, 
int(t.size()) - 1); i > 0; i -= i & -i) s += t[i]; return s; }

13 void dfsL(int u, int p) { up[u][0] = p; for (int k = 1; k < 17; k++) up[u][k] = up[up[u][k - 1]][k - 1]; 
for (int k = 0; k < int(g[u].size()); k++) { int v = g[u][k]; if (v != p) { dep[v] = dep[u] + 1; dfsL(v, 
u); } } }
14 int lca(int a, int b) {
15     if (dep[a] < dep[b]) swap(a, b);
16     for (int k = 16; k >= 0; k--) if (dep[a] - (1 << k) >= dep[b]) a = up[a][k];
17     if (a == b) return a;
18     for (int k = 16; k >= 0; k--) if (up[a][k] != up[b][k]) { a = up[a][k]; b = up[b][k]; }
19     return up[a][0];
20 }
21 int dist(int a, int b) { return dep[a] + dep[b] - 2 * dep[lca(a, b)]; }
22 int calcSize(int u, int p) { sz[u] = 1; for (int k = 0; k < int(g[u].size()); k++) { int v = g[u][k]; if 
(v != p && !removed[v]) sz[u] += calcSize(v, u); } return sz[u]; }
23 int centroid(int u, int p, int total) {
24     for (int k = 0; k < int(g[u].size()); k++) { int v = g[u][k]; if (v != p && !removed[v] && sz[v] * 2 
> total) return centroid(v, u, total); }
25     return u;
26 }
27 int build(int u, int parentC) {
28     int total = calcSize(u, 0), c = centroid(u, 0, total);
29     cpar[c] = parentC; removed[c] = true;
30     F[c].assign(total + 2, 0); Gt[c].assign(total + 2, 0);
31     for (int k = 0; k < int(g[c].size()); k++) { int v = g[c][k]; if (!removed[v]) build(v, c); }
32     return c;
33 }
34 void change(int x, long long v) {                // add v at vertex x
35     addF(F[x], 0, v);
36     for (int c = x; cpar[c]; c = cpar[c]) {
37         int p = cpar[c], d = dist(x, p);
38         addF(F[p], d, v); addF(Gt[c], d, v);
39     }
40 }
41 long long query(int x, int k) {
42     long long res = sumF(F[x], k);
43     for (int c = x; cpar[c]; c = cpar[c]) {
44         int p = cpar[c], d = dist(x, p);
45         res += sumF(F[p], k - d) - sumF(Gt[c], k - d);   // remove what c's piece already counted
46     }
47     return res;
48 }

12.4 莫队二次离线与 	Top	Tree
Prefix	contributions	offline;	top	trees	in	brief.
IDEA Second-offline	Mo ( 莫队二次离线 ): when moving a Mo pointer costs O(log n) (e.g. "count pairs i < j in
the window with ai > aj", 莫队二次离线  P4887), write each pointer move's contribution as f(prefix, x):
"the effect of element x against the prefix 1 … p". Record all those (p, x-range) requests offline, sweep p
from 1 to n once with a structure that makes each request O(1), and add the results back. Total O(n√n)
instead of O(n√n log n).
KEY POINT Top	trees (Top Tree) maintain path and subtree information on a dynamic forest at once, generalising LCT
( 拓展篇  第 14 章 ); they are almost never required in contests: LCT plus "virtual subtree" sums (keeping the
sum of non-preferred children) covers subtree queries in practice.
12.5 zkw	 线段树与线段树分裂
Bottom-up	loops;	move	nodes	into	a	new	tree.
IDEA zkw	segment	tree (zkw 线段树 , bottom-up): put the leaves at positions M + i where M is a power of two
≥ n + 2, parents at i / 2. A point update walks up with p >>= 1. A range query [l, r] starts from the open
ends l − 1 and r + 1 and moves both up until they become siblings, adding the sibling subtrees they pass.

IDEA Segment	tree	splitting ( 线段树分裂 ) is the reverse of merging ( 拓展篇 ): to move the values in [x, y] out
of a dynamic (value-indexed) segment tree into a new tree, walk down both at once. A node fully inside [x,

10 void add(int &o, int l, int r, int x, ll q) {
11     if (!o) o = ++cnt;
12     s[o] += q;
13     if (l == r) return;
14     int md = (l + r) / 2;
15     if (x <= md) add(lc[o], l, md, x, q); else add(rc[o], md + 1, r, x, q);
16 }
17 int merge(int a, int b, int l, int r) {
18     if (!a || !b) return a | b;
19     s[a] += s[b];
20     if (l == r) return a;
21     int md = (l + r) / 2;
22     lc[a] = merge(lc[a], lc[b], l, md); rc[a] = merge(rc[a], rc[b], md + 1, r);
23     return a;
24 }
25 void split(int &a, int &b, int l, int r, int x, int y) {     // move [x, y] of tree a into tree b
26     if (!a || y < l || r < x) return;
27     if (x <= l && r <= y) { b = a; a = 0; return; }           // whole node moves
28     if (!b) b = ++cnt;
29     int md = (l + r) / 2;
30     split(lc[a], lc[b], l, md, x, y); split(rc[a], rc[b], md + 1, r, x, y);
31     s[a] = s[lc[a]] + s[rc[a]]; s[b] = s[lc[b]] + s[rc[b]];
32 }
33 ll query(int o, int l, int r, int x, int y) {
34     if (!o || y < l || r < x) return 0;
35     if (x <= l && r <= y) return s[o];

37     return query(lc[o], l, md, x, y) + query(rc[o], md + 1, r, x, y);

43     return kth(rc[o], md + 1, r, k - s[lc[o]]);
44 }

12.6 珂朵莉树
Segments	in	a	set;	split,	assign,	walk.
IDEA The Chtholly	tree ( 珂朵莉树 , ODT, 颜色段均摊 ) stores the array as a set of maximal segments [l, r] with
one value. split(pos) cuts the segment containing pos so that a new one starts at pos. Range assign =
split(r + 1), split(l), erase everything between, insert one segment. Other operations just walk the
segments in [l, r].
KEY POINT Fast only when random	range	assignments keep the number of segments small (expected O(n log log n)
total, CF896C). Any operation that walks segments costs their count, so without assignments, or on
adversarial data, it degrades to O(n) per operation. Always split(r + 1) before split(l): the other order can
invalidate the first iterator.
