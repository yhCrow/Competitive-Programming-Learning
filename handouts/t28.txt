28.1 替罪羊树
Rotations	vs	rebuilding;	α-balance;	lazy	deletion.
IDEA Balanced BSTs differ in how they stay balanced. A rotation ( 旋转 ) changes the parent–child relation of
two nodes while keeping the in-order sequence: rotating x up over its parent y makes y a child of x and
hands x's inner subtree to y. Rotating Treaps and Splay trees use rotations; FHQ Treap ( 第 4 章 ) uses split /
merge; the scapegoat tree uses complete rebuilding.

IDEA Scapegoat	tree: choose α ≈ 0.7. After an insert or delete, walk back up; if some node x has a child subtree
larger than α·size(x) (or too many deleted nodes), it is the "scapegoat": collect its subtree in order and
rebuild it as a perfectly balanced tree. Deletion only lowers a counter (lazy), so deleted nodes are dropped
at the next rebuild.
KEY POINT A rebuild of a subtree of size s costs O(s), but it happens only after Ω(s) updates touched that subtree, so
the amortised cost per operation is O(log n). The code is plain recursion with no rotations.
竞 赛 拓 展 专 题  · 拓 展 篇  · 数 据 结 构 补 充
Rebuild unbalanced subtrees instead of rotating (scapegoat), move every accessed node to the root (Splay),
and search points in k dimensions with bounding boxes (K-D tree).

1 #include <bits/stdc++.h>
2 using namespace std;

4 const int N = 200005;
5 const double ALPHA = 0.7;
6 int ls[N], rs[N], val[N], cnt[N], s[N], sz[N], sd[N], tot = 0, root = 0;
7 // s: nodes in subtree, sz: copies of values, sd: nodes with cnt > 0
8 vector<int> buf;
9  
10 void up(int x) {
11     s[x] = s[ls[x]] + s[rs[x]] + 1;
12     sz[x] = sz[ls[x]] + sz[rs[x]] + cnt[x];
13     sd[x] = sd[ls[x]] + sd[rs[x]] + (cnt[x] > 0);
14 }
15 bool needRebuild(int x) {
16     return cnt[x] && (ALPHA * s[x] <= max(s[ls[x]], s[rs[x]]) || ALPHA * s[x] >= sd[x]);
17 }
18 void flatten(int x) {                               // in-order, keep only live nodes
19     if (!x) return;
20     flatten(ls[x]);
21     if (cnt[x]) buf.push_back(x);
22     flatten(rs[x]);
23 }
24 int build(int l, int r) {                           // perfectly balanced from buf[l..r]
25     if (l > r) return 0;
26     int mid = (l + r) / 2, x = buf[mid];
27     ls[x] = build(l, mid - 1); rs[x] = build(mid + 1, r);
28     up(x); return x;
29 }
30 void rebuild(int &x) { buf.clear(); flatten(x); x = build(0, int(buf.size()) - 1); }
31  
32 void insert(int &x, int v) {
33     if (!x) { x = ++tot; val[x] = v; cnt[x] = 1; up(x); return; }
34     if (v == val[x]) cnt[x]++;
35     else if (v < val[x]) insert(ls[x], v);
36     else insert(rs[x], v);
37     up(x);
38     if (needRebuild(x)) rebuild(x);
39 }
40 void remove_(int &x, int v) {
41     if (v == val[x]) cnt[x]--;                      // lazy: the node stays
42     else if (v < val[x]) remove_(ls[x], v);
43     else remove_(rs[x], v);
44     up(x);
45     if (needRebuild(x)) rebuild(x);
46 }
47 int countLess(int x, int v) {
48     if (!x) return 0;
49     if (v <= val[x]) return countLess(ls[x], v);
50     return sz[ls[x]] + cnt[x] + countLess(rs[x], v);
51 }
52 int kth(int x, int k) {
53     while (true) {
54         if (k <= sz[ls[x]]) x = ls[x];
55         else if (k <= sz[ls[x]] + cnt[x]) return val[x];
56         else { k -= sz[ls[x]] + cnt[x]; x = rs[x]; }
57     }
58 }

28.2 Splay	 树
Zig-zig	and	zig-zag;	splay	after	every	access.
IDEA Splay	tree ( 伸展树 ): after every access, splay the node to the root with rotations. When x, its parent and
grandparent lie in a straight line ("zig-zig") rotate the parent first, then x; when they zig-zag, rotate x twice.
That rule is what gives the amortised O(log n); always rotating x alone does not.
KEY POINT Because the accessed node becomes the root, operations become simple: to delete v, splay it to the root
and join its two subtrees; to cut a range [l, r] of a sequence, splay l − 1 to the root and r + 1 below it, and
the range is one subtree. LCT ( 第 14 章 ) is built on splay trees.

splay.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 const int N = 200005;
5 int rt = 0, tot = 0, fa[N], ch[N][2], val[N], cnt[N], sz[N];
6  
7 void up(int x) { sz[x] = sz[ch[x][0]] + sz[ch[x][1]] + cnt[x]; }
8 bool side(int x) { return ch[fa[x]][1] == x; }
9 void rotate(int x) {
10     int y = fa[x], z = fa[y], k = side(x);
11     ch[y][k] = ch[x][!k]; if (ch[x][!k]) fa[ch[x][!k]] = y;
12     ch[x][!k] = y; fa[y] = x; fa[x] = z;
13     if (z) ch[z][ch[z][1] == y] = x;
14     up(y); up(x);
15 }
16 void splay(int x) {                                 // bring x to the root
17     for (int f; (f = fa[x]) != 0; rotate(x))
18         if (fa[f]) rotate(side(x) == side(f) ? f : x);   // zig-zig: parent first
19     rt = x;
20 }
21 void insert(int v) {
22     if (!rt) { rt = ++tot; val[rt] = v; cnt[rt] = sz[rt] = 1; return; }
23     int x = rt, f = 0;
24     while (x && val[x] != v) { f = x; x = ch[x][v > val[x]]; }
25     if (x) cnt[x]++;
26     else { x = ++tot; val[x] = v; cnt[x] = 1; fa[x] = f; ch[f][v > val[f]] = x; }
27     up(x); splay(x);
28 }
29 int countLess(int v) {                              // values < v, splays the last node visited
30     int x = rt, res = 0, last = 0;
31     while (x) {
32         last = x;
33         if (v <= val[x]) x = ch[x][0];
34         else { res += sz[ch[x][0]] + cnt[x]; x = ch[x][1]; }
35     }
36     if (last) splay(last);
37     return res;
38 }
39 int kth(int k) {
40     int x = rt;
41     while (true) {
42         if (k <= sz[ch[x][0]]) x = ch[x][0];
43         else if (k <= sz[ch[x][0]] + cnt[x]) { splay(x); return val[x]; }
44         else { k -= sz[ch[x][0]] + cnt[x]; x = ch[x][1]; }
45     }
46 }
47 void remove_(int v) {
48     int x = rt;
49     while (val[x] != v) x = ch[x][v > val[x]];
50     splay(x);
51     if (cnt[x] > 1) { cnt[x]--; up(x); return; }
52     int L = ch[x][0], R = ch[x][1];
53     fa[L] = fa[R] = 0;
54     if (!L) { rt = R; return; }
55     int p = L;
56     while (ch[p][1]) p = ch[p][1];                  // largest of the left part
57     splay(p);                                       // now root of the left part, no right child
58     ch[p][1] = R; if (R) fa[R] = p;
59     up(p);
60 }

28.3 K-D	 树
Alternating	medians;	boxes;	nearest	neighbour.
IDEA A K-D	tree (K-D 树 ) organises points in k dimensions: the root splits the points by the median x-
coordinate, its children by the median y, and so on alternating. Each node also stores the bounding	box of
its subtree. Building with nth_element is O(n log n).
KEY POINT Nearest neighbour: visit the subtree on the query's side first, and skip any subtree whose bounding box is
already farther than the best distance found. Rectangle queries: take whole subtrees whose box is inside,
skip boxes outside. Worst case for a 2D rectangle query is O(√n) ( 简单题  P4148, 天使玩偶  P4169).

1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 const int N = 200005;
5 struct P { long long c[2]; };
6 P p[N];
7 int lc[N], rc[N];
8 long long lo[N][2], hi[N][2], best;
9 int K;                                        // current split dimension (used by the comparator)
10  
11 bool cmpK(const P &a, const P &b) { return a.c[K] < b.c[K]; }
12 int build(int l, int r, int dim) {            // node = index of the median point
13     if (l > r) return -1;
14     int mid = (l + r) / 2;
15     K = dim;
16     nth_element(p + l, p + mid, p + r + 1, cmpK);
17     lc[mid] = build(l, mid - 1, dim ^ 1);
18     rc[mid] = build(mid + 1, r, dim ^ 1);
19     for (int d = 0; d < 2; d++) {
20         lo[mid][d] = hi[mid][d] = p[mid].c[d];
21         if (lc[mid] >= 0) { lo[mid][d] = min(lo[mid][d], lo[lc[mid]][d]); hi[mid][d] = max(hi[mid][d], 
hi[lc[mid]][d]); }
22         if (rc[mid] >= 0) { lo[mid][d] = min(lo[mid][d], lo[rc[mid]][d]); hi[mid][d] = max(hi[mid][d], 
hi[rc[mid]][d]); }
23     }
24     return mid;
25 }
26 long long boxDist(int t, const P &q) {        // squared distance from q to t's bounding box
27     long long s = 0;
28     for (int d = 0; d < 2; d++) {
29         long long g = 0;
30         if (q.c[d] < lo[t][d]) g = lo[t][d] - q.c[d];
31         else if (q.c[d] > hi[t][d]) g = q.c[d] - hi[t][d];
32         s += g * g;
33     }
34     return s;
35 }
36 void query(int t, const P &q) {
37     if (t < 0 || boxDist(t, q) >= best) return;              // the whole box is too far
38     long long dx = p[t].c[0] - q.c[0], dy = p[t].c[1] - q.c[1];
39     best = min(best, dx * dx + dy * dy);
40     long long dl = lc[t] >= 0 ? boxDist(lc[t], q) : LLONG_MAX, dr = rc[t] >= 0 ? boxDist(rc[t], q) : 
LLONG_MAX;
41     if (dl < dr) { query(lc[t], q); query(rc[t], q); }        // nearer box first
42     else { query(rc[t], q); query(lc[t], q); }
43 }

45 int main() {
46     int n, m;
47     cin >> n >> m;

52         best = LLONG_MAX;

55     }

Key	points	to	memorise
1. Rotation: x up over its parent; in-order unchanged.
2. Scapegoat: rebuild a subtree when a child > α·size or too many deleted; amortised O(log n).
3. Lazy delete: cnt − 1, drop dead nodes at the next rebuild.
4. Splay: zig-zig rotate parent then x; zig-zag rotate x twice; splay every access.
5. Delete in splay: splay v, join left (max splayed up) and right.
6. K-D tree: alternate split dimensions, store boxes, prune boxes farther than best.
