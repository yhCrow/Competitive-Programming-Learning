5.1 树上主席树
Root-path	versions;	u	+	v	−	lca	−	parent(lca).
IDEA Persistent	segment	trees	on	a	tree ( 树上主席树 ): give every vertex the value tree of its whole root
path, built from its parent's version plus its own value ( 进阶篇  第 8 章  did the same along an array). Then
the multiset of values on the path u–v is version(u) + version(v) − version(lca) − version(parent(lca)),
and the k-th smallest is found by walking those four trees together (Count on a tree, P2633).
KEY POINT Memory n log n nodes; each query O(log n) after an LCA. Forests that merge ( 森林  P3302) rebuild the
smaller tree's versions on top of the other one (small-to-large).
竞 赛 拓 展 专 题  · NOI 篇  · NOI 专 题
Path k-th smallest with persistent trees on root paths, range chmin with sums by keeping the top two
values, and depth-indexed tree DP in O(n) by reusing the long child's memory.

15     up[u][0] = p;

20 int lca(int x, int y) {
21     if (dep[x] < dep[y]) swap(x, y);
22     for (int k = LOG - 1; k >= 0; k--) if (dep[x] - (1 << k) >= dep[y]) x = up[x][k];
23     if (x == y) return x;
24     for (int k = LOG - 1; k >= 0; k--) if (up[x][k] != up[y][k]) { x = up[x][k]; y = up[y][k]; }
25     return up[x][0];
26 }

30     if (k <= c) return kth(ls[u], ls[v], ls[w], ls[z], l, mid, k);
31     return kth(rs[u], rs[v], rs[w], rs[z], mid + 1, r, k - c);
32 }

5.2 吉司机线段树
Max,	strict	second	max,	count;	amortised	chmin.
IDEA Segment	tree	beats ( 吉司机线段树 ) supports the "range chmin" update ai = min(ai, x) together with
range sums. Each node stores the maximum mx, the strict	second	maximum se, and how many times
the maximum occurs, cnt. For an update with x: if x ≥ mx, nothing changes; if se < x < mx, only the
maximums change, so sum −= cnt·(mx − x) and mx = x right here (a lazy tag); otherwise recurse into the
children.
KEY POINT The "otherwise" case looks slow, but it merges the two largest distinct values of the node, which can only
happen O(log n) times per value: amortised O((n + q) log n) for chmin with sums. Historic maxima (CPU
监控  P4314, P6242) add more tags on the same idea.

6 long long a[N], mx[N * 4], se[N * 4], cntmx[N * 4], sum[N * 4];
7  
8 void pull(int p) {
9     int L = 2 * p, R = 2 * p + 1;
10     sum[p] = sum[L] + sum[R];
11     if (mx[L] == mx[R]) { mx[p] = mx[L]; cntmx[p] = cntmx[L] + cntmx[R]; se[p] = max(se[L], se[R]); }
12     else if (mx[L] > mx[R]) { mx[p] = mx[L]; cntmx[p] = cntmx[L]; se[p] = max(se[L], mx[R]); }
13     else { mx[p] = mx[R]; cntmx[p] = cntmx[R]; se[p] = max(mx[L], se[R]); }
14 }
15 void applyMin(int p, long long x) {                  // only the maximums are above x
16     if (x >= mx[p]) return;
17     sum[p] -= cntmx[p] * (mx[p] - x); mx[p] = x;
18 }
19 void push(int p) { applyMin(2 * p, mx[p]); applyMin(2 * p + 1, mx[p]); }
20 void build(int p, int l, int r) {
21     if (l == r) { mx[p] = sum[p] = a[l]; se[p] = LLONG_MIN; cntmx[p] = 1; return; }
22     int mid = (l + r) / 2; build(2 * p, l, mid); build(2 * p + 1, mid + 1, r); pull(p);
23 }
24 void chmin(int p, int l, int r, int x, int y, long long v) {
25     if (y < l || r < x || v >= mx[p]) return;          // nothing to change
26     if (x <= l && r <= y && v > se[p]) { applyMin(p, v); return; }   // only the max changes: tag here
27     push(p);
28     int mid = (l + r) / 2;
29     chmin(2 * p, l, mid, x, y, v); chmin(2 * p + 1, mid + 1, r, x, y, v);
30     pull(p);
31 }
32 long long qsum(int p, int l, int r, int x, int y) {
33     if (y < l || r < x) return 0;
34     if (x <= l && r <= y) return sum[p];
35     push(p); int mid = (l + r) / 2;
36     return qsum(2 * p, l, mid, x, y) + qsum(2 * p + 1, mid + 1, r, x, y);
37 }
38 long long qmax(int p, int l, int r, int x, int y) {
39     if (y < l || r < x) return LLONG_MIN;
40     if (x <= l && r <= y) return mx[p];
41     push(p); int mid = (l + r) / 2;
42     return max(qmax(2 * p, l, mid, x, y), qmax(2 * p + 1, mid + 1, r, x, y));

5.3 长链剖分
Deepest	child;	shifted	arrays;	O(n)	merges.
IDEA Long-path	decomposition ( 长链剖分 ): like heavy-light, but the "long" child is the one with the deepest
subtree. For DP indexed by depth (f[u][j] = number of vertices j levels below u), the long child's array can
be reused by shifting it one place (f[u][j + 1] = f[long][j]) with a shared memory pool, and only short
children are merged element by element.
KEY POINT Every vertex is the top of exactly one long chain, and a short child's array has length = its chain length, so
all merges together cost O(n). Uses: depth-indexed tree DP (CF1009F Dominant Indices), k-th ancestor in
O(1) after O(n log n) preprocessing (P5903).

6 int n, len_[N], son[N], ans[N];
7 int pool[N], *f[N], *ptr = pool;                    // shared memory: long chains are contiguous

9 void dfs1(int u, int p) {
10     for (int k = 0; k < int(g[u].size()); k++) {
11         int v = g[u][k]; if (v == p) continue;
12         dfs1(v, u);
13         if (len_[v] > len_[son[u]]) son[u] = v;
14     }
15     len_[u] = len_[son[u]] + 1;                     // vertices on the longest downward chain
16 }
17 void dfs2(int u, int p) {
18     f[u][0] = 1; ans[u] = 0;
19     if (son[u]) {
20         f[son[u]] = f[u] + 1;                       // long child: shifted view of the same memory
21         dfs2(son[u], u);
22         if (f[u][ans[son[u]] + 1] > f[u][ans[u]]) ans[u] = ans[son[u]] + 1;
23     }
24     for (int k = 0; k < int(g[u].size()); k++) {
25         int v = g[u][k]; if (v == p || v == son[u]) continue;
26         f[v] = ptr; ptr += len_[v];                 // short child: its own block
27         dfs2(v, u);
28         for (int j = 0; j < len_[v]; j++) {         // merge, cost = short chain length
29             f[u][j + 1] += f[v][j];
30             if (f[u][j + 1] > f[u][ans[u]] || (f[u][j + 1] == f[u][ans[u]] && j + 1 < ans[u])) ans[u] = 
j + 1;
31         }
32     }
33     if (f[u][ans[u]] == 1) ans[u] = 0;              // all depths have one vertex: smallest depth 0
34 }

KEY POINT For every vertex: the smallest depth j (below it) with the most vertices at that depth in its subtree. Checked

5.4 左偏树
dist;	merge	down	right	spines;	delete-min	with	DSU.
IDEA A leftist	heap ( 左偏树 , a mergeable heap 可并堆 ) is a binary heap-ordered tree where each node stores
dist = length of the shortest path down to a missing child. The rule dist(left)	≥	dist(right) keeps the right
spine O(log n) long, so two heaps merge in O(log n) by walking down right spines: keep the smaller root,
merge its right child with the other heap, swap children if the rule breaks.
KEY POINT Push = merge with a one-node heap; pop = merge the two children of the root. Together with a DSU that
maps each element to its heap root this solves "merge the heaps of x and y / delete the minimum of x's
heap" (P3377). Its persistent version (copy nodes on the merge path) gives the fast k-th shortest path
algorithm (NOI 篇  第 11 章 ).

8 int merge(int x, int y) {
9     if (!x || !y) return x | y;

13     dis[x] = dis[rs[x]] + 1;
14     return x;
15 }

5.5 边分治
Binarise;	split	at	the	most	balanced	edge.
IDEA Edge	divide	and	conquer ( 边分治 ): like centroid decomposition ( 点分治 ), but split the tree at the edge
whose removal leaves the most balanced two parts. Every path through that edge is "a path in the left part
ending at its end + the edge + a path in the right part", so each step combines exactly two sides, which is
easier than merging many subtrees of a centroid.
KEY POINT A star breaks it (every edge leaves a part of size n − 1), so first binarise the tree: replace each vertex's
children by a chain of virtual vertices joined by weight-0 edges. Degrees become ≤ 3 and the best edge
always leaves parts of size ≥ about n/3, so the depth is O(log n). Virtual vertices are not counted. Used
when the two-sided merge matters (P4220 通道 , and many "maximise f(path) over pairs" problems).

5.6 课后习题与实验

Key	points	to	memorise
