23.1 线段树优化建图
Out-tree	down,	in-tree	up;	ranges	=	O(log	n)	nodes.
IDEA Segment-tree	graph	building ( 线段树优化建图 ): edges "from u to every vertex in [l, r]" (or the reverse)
would cost O(n) edges each. Build two segment trees whose nodes are extra vertices: an out-tree with
edges parent → child (weight 0), whose leaves are the real vertices, and an in-tree with edges child →
parent (weight 0). Then u → [l, r] is u → the O(log n) covering nodes of the out-tree; [l, r] → u goes from
the covering nodes of the in-tree to u.
KEY POINT O((n + m log n)) edges in total, then any graph algorithm runs as usual (Dijkstra here, CF786B Legacy; also
SCC and 2-SAT). The leaves of both trees must be the same real vertices (or joined by 0-edges), so paths
can enter one tree and leave the other.
竞 赛 拓 展 专 题  · NOI 篇  · NOI 专 题
Replace range edges by O(log n) edges into segment-tree nodes, shrink "at most one" 2-SAT groups with
prefix variables, and get all-pairs shortest paths with negative edges by reweighting.

4 typedef long long ll;
5 int n, q, s, idOut[400005], idIn[400005], cntV;
6 vector<pair<int, ll>> g[2000005];
7  
8 void build(int p, int l, int r) {                 // node ids: real vertices are 1..n
9     if (l == r) { idOut[p] = idIn[p] = l; return; }
10     idOut[p] = ++cntV; idIn[p] = ++cntV;
11     int mid = (l + r) / 2;
12     build(2 * p, l, mid); build(2 * p + 1, mid + 1, r);
13     g[idOut[p]].push_back(make_pair(idOut[2 * p], 0LL)); g[idOut[p]].push_back(make_pair(idOut[2 * p + 
1], 0LL));   // out-tree: down
14     g[idIn[2 * p]].push_back(make_pair(idIn[p], 0LL)); g[idIn[2 * p + 1]].push_back(make_pair(idIn[p], 
0LL));       // in-tree: up
15 }
16 void link(int p, int l, int r, int x, int y, int u, ll w, bool toRange) {
17     if (y < l || r < x) return;
18     if (x <= l && r <= y) {
19         if (toRange) g[u].push_back(make_pair(idOut[p], w));    // u -> whole node
20         else g[idIn[p]].push_back(make_pair(u, w));             // whole node -> u
21         return;
22     }
23     int mid = (l + r) / 2;
24     link(2 * p, l, mid, x, y, u, w, toRange); link(2 * p + 1, mid + 1, r, x, y, u, w, toRange);
25 }

45         }
46     }

23.2 2-SAT	 前缀优化
Prefix	variables	P_i;	O(k)	clauses	per	group.
IDEA 2-SAT	with	"at	most	one"	groups: forbidding every pair inside a group of size k costs O(k²) clauses. Add
prefix	variables Pi = "some of the first i members is true". Clauses: xi → Pi, Pi−1 → Pi, and Pi−1 → ¬xi (if an
earlier member is true, xi must be false). O(k) clauses, the same answers (Riddle P6378).
KEY POINT The same prefix trick (or a segment tree over the literals, 23.1) shrinks any "range of implications" in 2-
SAT. Solve with SCCs as in 进阶篇  第 12 章 : unsatisfiable iff some x and ¬x share an SCC.

1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 int n, m, k, V, dfn[200005], low[200005], scc[200005], timer_ = 0, sccCnt = 0;
5 vector<int> g[200005], stk;
6 bool onStk[200005];
7 int lit(int v, bool val) { return 2 * v + (val ? 0 : 1); }        // node for "v == val"
8 void imply(int a, int b) { g[a].push_back(b); g[b ^ 1].push_back(a ^ 1); }   // a -> b and not b -> not 
a
9 void tarjan(int u) {
10     dfn[u] = low[u] = ++timer_; stk.push_back(u); onStk[u] = true;
11     for (int i = 0; i < int(g[u].size()); i++) {
12         int v = g[u][i];
13         if (!dfn[v]) { tarjan(v); low[u] = min(low[u], low[v]); }
14         else if (onStk[v]) low[u] = min(low[u], dfn[v]);
15     }
16     if (low[u] == dfn[u]) { ++sccCnt; while (true) { int x = stk.back(); stk.pop_back(); onStk[x] = 
false; scc[x] = sccCnt; if (x == u) break; } }
17 }

27         int a = cls[i].first, b = cls[i].second;
28         imply(lit(abs(a), a < 0), lit(abs(b), b > 0));

false)); }

37         }
38     }

23.3 Johnson	 全源最短路
Potentials	from	Bellman-Ford;	Dijkstra	n	times.
IDEA Johnson's	algorithm (Johnson 全源最短路 ) finds all-pairs shortest paths with negative edges (but no
negative cycle) in O(nm log m), better than Floyd's O(n³) on sparse graphs. Add a virtual vertex 0 with 0-
edges to all vertices; Bellman-Ford / SPFA gives potentials h(v) = dist(0, v). Reweight w'(u, v) = w(u, v) +
h(u) − h(v) ≥ 0, run Dijkstra from every vertex, and undo: dist(u, v) = dist'(u, v) − h(u) + h(v).
KEY POINT Why reweighting keeps shortest paths: along any path u → v the potentials telescope, so every path's
length changes by the same h(u) − h(v). And w' ≥ 0 because h(v) ≤ h(u) + w(u, v) (the triangle inequality
of shortest distances). A negative cycle shows up in the Bellman-Ford step (P5905).

johnson.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 typedef long long ll;
5 const ll INF = LLONG_MAX / 4;
6 int main() {
7     int n, m;
8     cin >> n >> m;
9     vector<array<ll, 3>> e(m);
10     vector<vector<pair<int, ll>>> g(n + 1);
11     for (int i = 0; i < m; i++) { cin >> e[i][0] >> e[i][1] >> e[i][2]; g[e[i]
[0]].push_back(make_pair(int(e[i][1]), e[i][2])); }
12     vector<ll> h(n + 1, 0);                        // Bellman-Ford from a virtual source: start all at 0
13     for (int it = 0; it < n; it++) {
14         bool changed = false;
15         for (int i = 0; i < m; i++) if (h[e[i][0]] + e[i][2] < h[e[i][1]]) { h[e[i][1]] = h[e[i][0]] + 
e[i][2]; changed = true; }
16         if (!changed) break;
17         if (it == n - 1) { cout << -1 << "\n"; return 0; }   // still improving: negative cycle
18     }
19     for (int s = 1; s <= n; s++) {
20         vector<ll> dist(n + 1, INF);
21         priority_queue<pair<ll, int>, vector<pair<ll, int>>, greater<pair<ll, int>>> pq;
22         dist[s] = 0; pq.push(make_pair(0LL, s));
23         while (!pq.empty()) {
24             ll dd = pq.top().first; int u = pq.top().second; pq.pop();
25             if (dd > dist[u]) continue;
26             for (int k = 0; k < int(g[u].size()); k++) {
27                 int v = g[u][k].first; ll w2 = g[u][k].second + h[u] - h[v];   // reweighted, >= 0
28                 if (dd + w2 < dist[v]) { dist[v] = dd + w2; pq.push(make_pair(dist[v], v)); }
29             }
30         }
31         for (int v = 1; v <= n; v++) {
32             if (dist[v] == INF) cout << "x"; else cout << dist[v] - h[s] + h[v];   // undo the 
potentials
33             cout << (v < n ? " " : "");
34         }
35         cout << "\n";
36     }
37     return 0;
38 }

The	Luogu	problems	for	this	chapter,	a	ladder	through	them,	and	experiments	to	try	yourself.
Key	points	to	memorise
1. Out-tree: parent → child 0-edges; u → range = u → covering nodes.
2. In-tree: child → parent 0-edges; range → u = covering nodes → u.
3. At most one per group: P_i ⇐ x_i, P_i ⇐ P_(i−1), P_(i−1) ⇒ ¬x_i.
4. Johnson: h from Bellman-Ford; w' = w + h(u) − h(v) ≥ 0; undo after Dijkstra.
