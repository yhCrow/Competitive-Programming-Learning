30.1 0/1	 分数规划
Binary	search	λ;	check	Σ(a	−	λb)	≥	0.
IDEA 0/1	fractional	programming (0/1 分数规划 ): choose items (xi ∈ {0, 1}) to maximise Σ aixi / Σ bixi.
Binary search the answer λ: the ratio can reach λ iff max Σ (ai − λ bi) xi ≥ 0. That check has no division, so
it is an ordinary greedy / knapsack / shortest-path problem.
KEY POINT Why: Σa / Σb ≥ λ ⇔ Σa − λΣb ≥ 0 (with Σb > 0). The check is monotone in λ, so binary search works.
Typical checks: "pick k items" (sort by a − λb), "weight at least W" (knapsack), "minimum mean cycle"
(negative cycle with weights b·λ − a, 最小圈  P3199).
竞 赛 拓 展 专 题  · 拓 展 篇  · 数 学 拓 展
Maximise a ratio by binary searching its value, convolve arithmetic functions over divisors, sum φ and μ up
to 2³¹ with Du's sieve, and merge congruences whose moduli share factors.

1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 int n, W, w[255], t[255];
5 double f[1005];
6  
7 bool feasible(double lam) {                   // can sum(t - lam*w) >= 0 with total weight >= W?
8     for (int j = 1; j <= W; j++) f[j] = -1e18;
9     f[0] = 0;
10     for (int i = 0; i < n; i++) {
11         double v = t[i] - lam * w[i];
12         for (int j = W; j >= 0; j--) {        // weight capped at W: "at least W"
13             int nj = min(W, j + w[i]);
14             if (f[j] > -1e17) f[nj] = max(f[nj], f[j] + v);
15         }
16     }
17     return f[W] >= 0;
18 }

21     cin >> n >> W;
22     for (int i = 0; i < n; i++) cin >> w[i] >> t[i];
23     double lo = 0, hi = 1001;

25         double mid = (lo + hi) / 2;
26         if (feasible(mid)) lo = mid; else hi = mid;
27     }

30.2 狄利克雷卷积
f	*	g	over	divisors;	Dirichlet	prefix	sums.
IDEA The Dirichlet	convolution ( 狄利克雷卷积 ) of two arithmetic functions: (f * g)(n) = Σd | n f(d) g(n/d). Key
identities, with 1(n) = 1, id(n) = n, ε(n) = [n = 1]: μ * 1 = ε (the μ property, 第 9 章 ), φ * 1 = id, μ * id = φ.
Multiplicative * multiplicative is multiplicative.

KEY POINT Dirichlet	prefix	sum ( 狄利克雷前缀和 ): b(n) = Σd | n a(d) for all n ≤ N in O(N log log N): for each prime
p, for i = 1 … N/p add a[i] into a[i·p]. It is SOS DP ( 第 12 章 ) over the exponents of primes (P5495).
dirichlet_prefix.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 int main() {
5     int n;
6     cin >> n;
7     vector<long long> a(n + 1);
8     for (int i = 1; i <= n; i++) cin >> a[i];
9     vector<bool> comp(n + 1, false);
10     for (int p = 2; p <= n; p++) {
11         if (comp[p]) continue;
12         for (int q = 2 * p; q <= n; q += p) comp[q] = true;     // simple sieve
13         for (int i = 1; i * p <= n; i++) a[i * p] += a[i];      // add along prime p, increasing i
14     }
15     for (int i = 1; i <= n; i++) cout << a[i] << (i < n ? " " : "");
16     cout << "\n";
17     return 0;
18 }

30.3 杜教筛
S(n)	from	H(n)	and	S(n/d);	memo	+	small	sieve.
IDEA Du's	sieve ( 杜教筛 ) computes S(n) = Σi≤n f(i) for n up to about 2³¹, faster than sieving to n. Pick g so that
h = f * g has an easy prefix sum. Then Σi≤n h(i) = Σd≤n g(d) S(⌊n/d⌋), and isolating d = 1: g(1)S(n) = H(n) −
Σd=2..n g(d) S(⌊n/d⌋).
KEY POINT For φ choose g = 1 (φ * 1 = id, H(n) = n(n+1)/2); for μ choose g = 1 (μ * 1 = ε, H(n) = 1). Use division
blocks for the sum, memoise S in a hash map, and sieve S directly for n up to about N2/3: total O(N2/3)
(P4213).

du_sieve.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 const int M = 2000000;                          // sieve directly below M (about N^(2/3))
5 long long sphi[M + 1], smu[M + 1];
6 int pr[M], cnt = 0;
7 bool comp[M + 1];
8 unordered_map<long long, long long> memoPhi, memoMu;
9  
10 long long Sphi(long long n) {
11     if (n <= M) return sphi[n];
12     if (memoPhi.count(n)) return memoPhi[n];
13     long long res = n * (n + 1) / 2;             // sum of id = sum of (phi * 1)
14     for (long long l = 2, r; l <= n; l = r + 1) {
15         r = n / (n / l);
16         res -= (r - l + 1) * Sphi(n / l);        // g = 1: subtract S(n/d) for d >= 2
17     }
18     return memoPhi[n] = res;
19 }
20 long long Smu(long long n) {
21     if (n <= M) return smu[n];
22     if (memoMu.count(n)) return memoMu[n];
23     long long res = 1;                           // sum of epsilon = 1
24     for (long long l = 2, r; l <= n; l = r + 1) {
25         r = n / (n / l);
26         res -= (r - l + 1) * Smu(n / l);
27     }
28     return memoMu[n] = res;
29 }
30  
31 int main() {
32     vector<long long> phi(M + 1), mu(M + 1);
33     phi[1] = mu[1] = 1;
34     for (int i = 2; i <= M; i++) {               // linear sieve
35         if (!comp[i]) { pr[cnt++] = i; phi[i] = i - 1; mu[i] = -1; }
36         for (int j = 0; j < cnt && 1LL * i * pr[j] <= M; j++) {
37             comp[i * pr[j]] = true;
38             if (i % pr[j] == 0) { phi[i * pr[j]] = phi[i] * pr[j]; mu[i * pr[j]] = 0; break; }
39             phi[i * pr[j]] = phi[i] * (pr[j] - 1); mu[i * pr[j]] = -mu[i];
40         }
41     }
42     for (int i = 1; i <= M; i++) { sphi[i] = sphi[i - 1] + phi[i]; smu[i] = smu[i - 1] + mu[i]; }
43     int T;
44     cin >> T;
45     while (T--) { long long n; cin >> n; cout << Sphi(n) << " " << Smu(n) << "\n"; }
46     return 0;
47 }

30.4 扩展中国剩余定理与威尔逊定理
Merging	with	gcd;	(p	−	1)!	≡	−1.
IDEA Extended	CRT ( 扩展中国剩余定理 ): merge x ≡ r1 (mod m1) and x ≡ r2 (mod m2) when the moduli are
not coprime. Write x = r1 + m1t; then m1t ≡ r2 − r1 (mod m2), a linear congruence ( 进阶篇  第 19 章 ):
solvable iff g = gcd(m1, m2) divides r2 − r1. The merged modulus is lcm(m1, m2).
KEY POINT Merge the equations one by one; products reach 10¹⁸, so multiply with __int128. Multi-variable linear
Diophantine equations a1x1 + … + akxk = c are solvable iff gcd(a1, …, ak) | c; solve them by combining two
coefficients at a time with exgcd.
excrt.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 typedef long long ll;
5 ll exgcd(ll a, ll b, ll &x, ll &y) {
6     if (!b) { x = 1; y = 0; return a; }
7     ll g = exgcd(b, a % b, y, x);
8     y -= a / b * x;
9     return g;
10 }
11  
12 int main() {
13     int n;
14     cin >> n;
15     ll M = 1, R = 0;                                  // x = R (mod M) so far
16     for (int i = 0; i < n; i++) {
17         ll m, r;
18         cin >> m >> r;                                // x = r (mod m)
19         ll t, k, g = exgcd(M, m, t, k);
20         ll diff = ((r - R) % m + m) % m;
21         if (diff % g) { cout << -1 << "\n"; return 0; }
22         ll mod = m / g;
23         t = ll(__int128(t % mod + mod) % mod * (diff / g) % mod);   // M t = diff (mod m)
24         R += M * t;
25         M = M / g * m;                                // lcm
26         R = (R % M + M) % M;
27     }
28     cout << R << "\n";
29     return 0;
30 }

KEY POINT Return −1 as soon as one merge is impossible. Keep R reduced modulo the current M after every merge.
KEY POINT Wilson's	theorem ( 威尔逊定理 ): p is prime ⇔ (p − 1)! ≡ −1 (mod p). It is a characterisation, not a fast
test, but it gives (p − 1)! mod p for free and helps simplify factorials modulo a prime, e.g. n! mod p with
the multiples of p removed (used inside exLucas).
30.5 课后习题与实验
The	Luogu	problems	for	this	chapter,	a	ladder	through	them,	and	experiments	to	try	yourself.
Key	points	to	memorise

6. Wilson: p prime ⇔ (p − 1)! ≡ −1 (mod p).
