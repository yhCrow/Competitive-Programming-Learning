1 #include <bits/stdc++.h>

4 // print every game's score; a game ends at >= target with a lead of 2
5 void play(const string &s, int target) {
6     int w = 0, l = 0;
7     for (int i = 0; i < int(s.size()); i++) {
8         char ch = s[i];
9         if (ch == 'W') w++;
10         else l++;
11         if (max(w, l) >= target && abs(w - l) >= 2) {
12             cout << w << ":" << l << "\n";
13             w = l = 0;                       // new game
14         }
15     }
16     cout << w << ":" << l << "\n";           // unfinished game, even 0:0
17 }
18

20     string s;
21     char ch;
22     while (cin >> ch && ch != 'E')           // ignore line breaks
23         if (ch == 'W' || ch == 'L') s += ch;
24     play(s, 11);
25     cout << "\n";
26     play(s, 21);
27     return 0;
28 }

toy_puzzle.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 int face[100005];
5 string name[100005];
6  
7 int main() {
8     int n, m;
9     cin >> n >> m;
10     for (int i = 0; i < n; i++) cin >> face[i] >> name[i];
11     int p = 0;
12     while (m--) {
13         int a, s;                    // a: 0 left, 1 right; s steps
14         cin >> a >> s;
15         if (face[p] == a) p = (p - s % n + n) % n;  // one way round
16         else p = (p + s) % n;                    // the other way
17     }
18     cout << name[p] << "\n";
19     return 0;
20 }

IDEA Number the seats 0 … n−1. Moving s seats round a circle is (p + s) % n  one way and (p - s + n) % n

KEY POINT Facing in or out flips left and right. Instead of four cases, notice that "facing direction equals instruction"
always means the same way round: one comparison decides it (P1563).

vector<int> toBig(const string &s) {

8.2.3 Multiplying	by	a	small	number,	and	factorials

1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 void mulSmall(vector<int> &d, int k) {   // d = d * k, k small
5     int carry = 0;
6     for (size_t i = 0; i < d.size(); i++) {
7         int t = d[i] * k + carry;
8         d[i] = t % 10;
9         carry = t / 10;
10     }
11     while (carry) { d.push_back(carry % 10); carry /= 10; }
12 }

Key	points	to	memorise
1. Simulation: list every rule, choose the state, write one step, know when to stop.
2. Check the output format character by character; most errors are misread rules.
3. Circle of n: (p	+	s)	%	n and (p	-	s	%	n	+	n)	%	n.
4. Rule tables go into a 2D array; repeating patterns use i	%	k.
5. Formatting: one small if per rule (sign, 1, x, power, first term).
6. Endless walks: cap the steps or detect a repeated state.
7. Big numbers: vector<int> of digits, ones	digit	first; read as a string.
8. Add: digit + digit + carry, keep % 10, pass / 10; loop while digits or	carry remain.
9. Compare by length, then from the top digit; subtract smaller from larger and add "−".
10. Borrow: +10 here, −1 at the next digit; always trim leading zeros.
11. Big × small: digit × k + carry; push the final carry digit by digit.
12. Big × big: c[i	+	j]	+=	a[i]	*	b[j], then one carry sweep.
13. Big ÷ small: from the top digit down, rem = rem × 10 + digit.
