13.1 KMP	 自动机
Transitions	for	every	character;	matrix	power.
IDEA The KMP	automaton (KMP 自动机 ) turns a pattern of length m into states 0 … m ("how many characters
of the pattern currently match") with a transition for every character: go[j][c] = j + 1 if p[j+1] = c, else
go[nxt[j]][c] (with go[0][c] = 0 on a mismatch). It is the one-pattern case of the AC automaton ( 拓展篇  第
2 章 ).
KEY POINT Counting strings of length n that avoid the pattern: DP over (length, state), never entering state m. When n
is huge (GT 考试  P3193: n up to 10⁹), the transition is the same every step, so build the (m × m) transition
matrix and raise it to the n-th power ( 进阶篇  第 22 章 ): O(m³ log n).
竞 赛 拓 展 专 题  · NOI 篇  · NOI 专 题
Count strings avoiding a pattern with the KMP automaton and matrix power, split a string into Lyndon
words with Duval's algorithm, and test tree isomorphism by hashing from the centroid.

kmp_automaton.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 int m, K;
5 struct Mat { long long a[25][25]; };
6 Mat mul(const Mat &x, const Mat &y) {
7     Mat r;
8     for (int i = 0; i < m; i++) for (int j = 0; j < m; j++) {
9         long long s = 0;
10         for (int k = 0; k < m; k++) s += x.a[i][k] * y.a[k][j];
11         r.a[i][j] = s % K;
12     }
13     return r;
14 }
15  
16 int main() {
17     long long n;
18     string p;
19     cin >> n >> m >> K >> p;
20     vector<int> nxt(m + 1, 0);
21     for (int i = 1, j = 0; i < m; i++) {              // KMP failure function (0-based pattern)
22         while (j > 0 && p[i] != p[j]) j = nxt[j];
23         if (p[i] == p[j]) j++;
24         nxt[i + 1] = j;
25     }
26     Mat T;
27     for (int i = 0; i < m; i++) for (int j = 0; j < m; j++) T.a[i][j] = 0;
28     for (int j = 0; j < m; j++)                       // from state j, try each digit
29         for (char c = '0'; c <= '9'; c++) {
30             int k = j;
31             while (k > 0 && p[k] != c) k = nxt[k];
32             if (p[k] == c) k++;
33             if (k < m) T.a[j][k]++;                   // state m (full match) is forbidden
34         }
35     Mat R;
36     for (int i = 0; i < m; i++) for (int j = 0; j < m; j++) R.a[i][j] = (i == j) % K;
37     for (; n > 0; n >>= 1, T = mul(T, T)) if (n & 1) R = mul(R, T);
38     long long ans = 0;
39     for (int j = 0; j < m; j++) ans += R.a[0][j];     // start in state 0, end anywhere safe
40     cout << ans % K << "\n";
41     return 0;
42 }

IDEA A Lyndon	word is strictly smaller than all of its proper suffixes (equivalently, than all its rotations), e.g.
"aab", "abb". Every string splits uniquely into Lyndon words w1 ≥ w2 ≥ … ≥ wk (Chen–Fox–Lyndon).
Duval's	algorithm finds this factorisation in O(n) with three pointers.
KEY POINT Duval: i = start of the current block, j = compare position, k = scan position. While s[k] ≥ s[j]: if equal,
advance both (a repeat of a Lyndon word is forming); if larger, j = i (the whole prefix becomes one longer
Lyndon word). When s[k] < s[j], output copies of length k − j and restart after them. The minimal rotation
of a string starts at the last Lyndon factor of s + s that begins in the first half (P1368 工艺 ).
duval.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 int main() {
5     string s;
6     cin >> s;
7     int n = s.size(), i = 0;
8     vector<int> ends;
9     while (i < n) {
10         int j = i, k = i + 1;
11         while (k < n && s[j] <= s[k]) {
12             if (s[j] < s[k]) j = i;                   // everything so far merges into one Lyndon word
13             else j++;                                 // equal: a periodic repeat continues
14             k++;
15         }
16         while (i <= j) { i += k - j; ends.push_back(i); }   // output whole periods of length k - j
17     }
18     for (int t = 0; t < int(ends.size()); t++) cout << ends[t] << (t + 1 < int(ends.size()) ? " " : "");
19     cout << "\n";
20     return 0;
21 }

13.3 树哈希与 	Ukkonen
Order-free	child	hashes;	centroid	roots.
IDEA Tree	hashing ( 树哈希 ) decides whether two trees are isomorphic. For a rooted tree, compute each
vertex's hash from the multiset of its children's hashes, in a way that does not depend on their order, e.g.
h(u) = 1 + Σ f(h(child)) with a fixed scrambling function f (xor-shift on 64-bit), or by sorting the children's
hashes.

KEY POINT For unrooted trees, root at the centroid (there are one or two): two trees are isomorphic iff their hashes
rooted at their centroids match (taking the smaller of two). A plain sum of child hashes collides easily; the
scrambling function f is what makes it reliable ( 树同构  P5043).

1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 typedef unsigned long long ull;
5 ull mix(ull x) {                                     // a fixed 64-bit scrambler (splitmix64 style)
6     x += 0x9e3779b97f4a7c15ULL;
7     x = (x ^ (x >> 30)) * 0xbf58476d1ce4e5b9ULL;
8     x = (x ^ (x >> 27)) * 0x94d049bb133111ebULL;
9     return x ^ (x >> 31);
10 }
11 vector<int> g[55];
12 int n, sz[55];
13  
14 ull hashRooted(int u, int p) {
15     ull h = 1;
16     for (int k = 0; k < int(g[u].size()); k++) { int v = g[u][k]; if (v != p) h += mix(hashRooted(v, 
u)); }
17     return h;
18 }
19 int sizes(int u, int p) { sz[u] = 1; for (int k = 0; k < int(g[u].size()); k++) { int v = g[u][k]; if (v 
!= p) sz[u] += sizes(v, u); } return sz[u]; }
20  
21 int main() {
22     int T;
23     cin >> T;
24     vector<ull> code(T);
25     for (int t = 0; t < T; t++) {
26         cin >> n;
27         for (int i = 1; i <= n; i++) g[i].clear();
28         for (int i = 2; i <= n; i++) { int p; cin >> p; g[i].push_back(p); g[p].push_back(i); }
29         sizes(1, 0);
30         ull best = ULLONG_MAX;
31         for (int u = 1; u <= n; u++) {               // centroids: largest piece <= n / 2
32             int big = n - sz[u];
33             for (int k = 0; k < int(g[u].size()); k++) { int v = g[u][k]; if (sz[v] < sz[u]) big = 
max(big, sz[v]); }
34             if (2 * big <= n) best = min(best, hashRooted(u, 0));
35         }
36         code[t] = best;
37         int first = t;
38         for (int s = 0; s < t; s++) if (code[s] == code[t]) { first = s; break; }
39         cout << first + 1 << "\n";                   // smallest index of an isomorphic tree
40     }
41     return 0;
42 }

KEY POINT Each tree is given by parents of vertices 2 … n. For each tree print the first earlier tree it is isomorphic to.
Checked in Python by the canonical bracket encoding (sorted child strings, minimised over all roots),

KEY POINT Ukkonen's	algorithm builds a suffix tree online in O(n) using an "active point" and suffix links between

13.4 序列自动机与二维哈希
nxt[i][c];	row	hash	then	column	hash.
IDEA The subsequence	automaton ( 序列自动机 ): nxt[i][c] = the first position j > i with sj = c (0 if none),
filled from the back in O(n · Σ). Testing whether t is a subsequence of s is then one jump per character,
O(|t|); it greedily takes the earliest match, which is always safe.
KEY POINT The automaton is a DAG whose paths from state 0 are exactly the distinct subsequences, so DP on it
counts them: f(i) = 1 + Σc f(nxt[i][c]), answer f(0) − 1 for nonempty ones. Common subsequences of two
strings = paths in both automata at once (P5826 is the template).

subseq_automaton.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 typedef long long ll;
5 const ll MOD = 1000000007;
6  
7 int main() {
8     string s;
9     int q;
10     cin >> s >> q;
11     int n = s.size();
12     vector<array<int, 26>> nxt(n + 1, array<int, 26>{});   // all 0 = no next occurrence
13     for (int i = n; i >= 1; i--) { nxt[i - 1] = nxt[i]; nxt[i - 1][s[i - 1] - 'a'] = i; }   // positions 
are 1-based
14     for (int k = 0; k < q; k++) {
15         string t;
16         cin >> t;
17         int p = 0;
18         bool ok = true;
19         for (int i = 0; i < int(t.size()) && ok; i++) { p = nxt[p][t[i] - 'a']; if (!p) ok = false; }
20         cout << (ok ? "Yes" : "No") << "\n";
21     }
22     vector<ll> f(n + 1, 1);                         // f[i] = distinct subsequences after state i, 
including empty
23     for (int i = n; i >= 0; i--)
24         for (int c = 0; c < 26; c++) if (nxt[i][c]) f[i] = (f[i] + f[nxt[i][c]]) % MOD;
25     cout << (f[0] - 1 + MOD) % MOD << "\n";
26     return 0;
27 }

KEY POINT Subsequence tests and the count of distinct nonempty subsequences, checked in Python by listing all 2¹²
subsequences of s.
IDEA 2D	hashing ( 二维哈希 ): hash each row's windows of width b with base P (unsigned overflow as mod
2⁶⁴), then hash each column of those row hashes over windows of height a with a second	base Q. A
matrix window then has one number, compared to the pattern's in O(1): counting occurrences of an a × b
pattern in an n × m text is O(nm).
KEY POINT Same rules as 1D hashing ( 拓展篇 ): two different bases for the two directions, and use two moduli or
random bases if the problem is adversarial. Prefix-sum style 2D hashes H(i, j) also allow any submatrix in
O(1): H = h(x2, y2) − h(x1, y2)·Qa − h(x2, y1)·Pb + h(x1, y1)·PbQa.

hash2d.cpp
1 #include <bits/stdc++.h>
2 using namespace std;
3  
4 typedef unsigned long long ull;
5 const ull P = 131, Q = 13331;
6  
7 int main() {
8     int n, m, a, b;
9     cin >> n >> m;
10     vector<string> t(n);
11     for (int i = 0; i < n; i++) cin >> t[i];
12     cin >> a >> b;
13     vector<string> p(a);
14     for (int i = 0; i < a; i++) cin >> p[i];
15     ull pb = 1, qa = 1;
16     for (int i = 0; i < b; i++) pb *= P;
17     for (int i = 0; i < a; i++) qa *= Q;
18     ull target = 0;                                 // hash of the pattern
19     for (int i = 0; i < a; i++) {
20         ull h = 0;
21         for (int j = 0; j < b; j++) h = h * P + p[i][j];
22         target = target * Q + h;
23     }
24     vector<vector<ull>> row(n, vector<ull>(m - b + 1));   // row[i][j] = hash of t[i][j..j+b-1]
25     for (int i = 0; i < n; i++) {
26         ull h = 0;
27         for (int j = 0; j < m; j++) {
28             h = h * P + t[i][j];
29             if (j >= b) h -= pb * t[i][j - b];
30             if (j >= b - 1) row[i][j - b + 1] = h;
31         }
32     }
33     int cnt = 0;
34     for (int j = 0; j + b <= m; j++) {              // slide down each column of row hashes
35         ull h = 0;
36         for (int i = 0; i < n; i++) {
37             h = h * Q + row[i][j];
38             if (i >= a) h -= qa * row[i - a][j];
39             if (i >= a - 1 && h == target) cnt++;
40         }
41     }
42     cout << cnt << "\n";
43     return 0;
44 }

13.5 课后习题与实验
The	Luogu	problems	for	this	chapter,	a	ladder	through	them,	and	experiments	to	try	yourself.
Key	points	to	memorise
1. KMP automaton: go[j][c] = j+1 on match, else go[nxt[j]][c]; avoid state m.
2. Fixed transitions over many steps ⇒ matrix power.
3. Lyndon: smaller than all proper suffixes; unique factorisation w1 ≥ w2 ≥ ….
4. Duval: s[k] > s[j] ⇒ j = i; equal ⇒ j++; smaller ⇒ output periods of length k − j.
5. Tree hash: h(u) = 1 + Σ mix(h(child)); unrooted ⇒ root at centroid(s).
6. Subsequence automaton: nxt[i][c] from the back; paths = distinct subsequences.
7. 2D hash: base P along rows, base Q down columns.
