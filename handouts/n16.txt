16.1 Berlekamp–Massey
Fix	the	recurrence	with	the	last	failure.
IDEA Berlekamp–Massey finds the shortest	linear	recurrence ai = Σj=1..L cj ai−j (mod a prime) that generates
a given sequence, in O(n²). Process terms one by one; when the current recurrence predicts ai wrongly by
a discrepancy Δ, correct it with a scaled, shifted copy of the recurrence that was current at the last failure.
With 2L terms of a sequence that truly has an order-L recurrence, the result is exact.
KEY POINT Contest use: when an answer sequence is known to satisfy some linear recurrence (DP with fixed
transitions, matrix powers, counting walks), compute the first few hundred terms by brute force, run BM,
then get the n-th term by matrix power or, faster, polynomial "Kitamasa" reduction (P5487).
竞 赛 拓 展 专 题  · NOI 篇  · NOI 专 题
Recover the shortest linear recurrence of a sequence from its first terms, and evaluate a polynomial at
many points with a product tree and polynomial remainders.

8 vector<ll> bm(const vector<ll> &s) {
9     vector<ll> cur, last;                       // cur: recurrence coefficients c_1..c_L
10     int lastFail = -1; ll lastDelta = 0;
11     for (int i = 0; i < int(s.size()); i++) {
12         ll t = 0;
13         for (int j = 0; j < int(cur.size()); j++) t = (t + cur[j] * s[i - 1 - j]) % MOD;
14         ll delta = (s[i] - t + MOD) % MOD;      // prediction error
15         if (delta == 0) continue;
16         if (lastFail < 0) { cur.assign(i + 1, 0); lastFail = i; lastDelta = delta; continue; }
17         ll coef = delta * power(lastDelta, MOD - 2) % MOD;
18         vector<ll> nxt(i - lastFail - 1, 0);    // shift of the old recurrence
19         nxt.push_back(coef);
20         for (int j = 0; j < int(last.size()); j++) nxt.push_back((MOD - coef) * last[j] % MOD);
21         if (nxt.size() < cur.size()) nxt.resize(cur.size(), 0);
22         for (int j = 0; j < int(cur.size()); j++) nxt[j] = (nxt[j] + cur[j]) % MOD;
23         if (i - lastFail + int(last.size()) >= int(cur.size())) { last = cur; lastFail = i; lastDelta = 
delta; }
24         cur = nxt;
25     }
26     return cur;
27 }

16.2 多项式多点求值
Product	tree;	remainders	down	the	tree.
IDEA Multipoint	evaluation ( 多项式多点求值 ): evaluate a degree-n polynomial A at n points in O(n log² n).
Build a product	tree: leaves (x − xi), each node the product of its children. Then go down: A mod (root
product), and at each node reduce the parent's remainder modulo the node's product. At leaf i the
remainder is the constant A(xi).
KEY POINT Polynomial division with remainder (O(n log n)): reverse A and B, multiply reversed A by the inverse
(NOI 篇  第 7 章 ) of reversed B modulo xdeg A − deg B + 1 to get the reversed quotient Q; then R = A − Q·B.
Below a small size, plain Horner evaluation is faster, which is the usual practical cut-off (P5050).

8 void ntt(Poly &a, bool inv) {
9     int n = a.size();
10     for (int i = 1, j = 0; i < n; i++) { int bit = n >> 1; for (; j & bit; bit >>= 1) j ^= bit; j ^= 
bit; if (i < j) swap(a[i], a[j]); }
11     for (int len = 2; len <= n; len <<= 1) {
12         ll w = power(G, (MOD - 1) / len); if (inv) w = power(w, MOD - 2);
13         for (int i = 0; i < n; i += len) { ll x = 1; for (int j = 0; j < len / 2; j++) {
14             ll u = a[i + j], v = a[i + j + len / 2] * x % MOD; a[i + j] = (u + v) % MOD; a[i + j + len / 
2] = (u - v + MOD) % MOD; x = x * w % MOD; } }
15     }
16     if (inv) { ll ni = power(n, MOD - 2); for (int i = 0; i < n; i++) a[i] = a[i] * ni % MOD; }
17 }
18 Poly mul(Poly a, Poly b) {
19     if (a.empty() || b.empty()) return Poly();
20     int need = a.size() + b.size() - 1, n = 1; while (n < need) n <<= 1;
21     a.resize(n); b.resize(n); ntt(a, false); ntt(b, false);
22     for (int i = 0; i < n; i++) a[i] = a[i] * b[i] % MOD;
23     ntt(a, true); a.resize(need); return a;
24 }
25 Poly inverse(const Poly &a, int n) {
26     Poly b(1, power(a[0], MOD - 2));
27     for (int k = 1; k < n; k <<= 1) {
28         Poly ak(a.begin(), a.begin() + min(int(a.size()), 2 * k));
29         Poly t = mul(ak, b); t.resize(2 * k);
30         for (int i = 0; i < 2 * k; i++) t[i] = (MOD - t[i]) % MOD;
31         t[0] = (t[0] + 2) % MOD;
32         b = mul(b, t); b.resize(2 * k);
33     }
34     b.resize(n); return b;
35 }
36 Poly modp(const Poly &a, const Poly &b) {            // a mod b
37     int n = a.size(), m = b.size();
38     if (n < m) return a;
39     Poly ra(a.rbegin(), a.rend()), rb(b.rbegin(), b.rend());
40     ra.resize(n - m + 1);
41     Poly q = mul(ra, inverse(rb, n - m + 1)); q.resize(n - m + 1);
42     reverse(q.begin(), q.end());
43     Poly qb = mul(q, b), r(m - 1);
44     for (int i = 0; i < m - 1; i++) r[i] = (a[i] - qb[i] + MOD) % MOD;
45     return r;
46 }
47 vector<Poly> tree_;
48 vector<ll> xs, ans;
49 void build(int p, int l, int r) {
50     if (l == r) { tree_[p] = Poly{(MOD - xs[l]) % MOD, 1}; return; }
51     int mid = (l + r) / 2;
52     build(2 * p, l, mid); build(2 * p + 1, mid + 1, r);
53     tree_[p] = mul(tree_[2 * p], tree_[2 * p + 1]);
54 }
55 void down(int p, int l, int r, const Poly &rem) {
56     Poly cur = modp(rem, tree_[p]);
57     if (l == r) { ans[l] = cur.empty() ? 0 : cur[0]; return; }
58     int mid = (l + r) / 2;
59     down(2 * p, l, mid, cur); down(2 * p + 1, mid + 1, r, cur);

16.3 课后习题与实验
The	Luogu	problems	for	this	chapter,	a	ladder	through	them,	and	experiments	to	try	yourself.
Key	points	to	memorise
1. BM: shortest recurrence; on a discrepancy, add a shifted, scaled copy of the last failing recurrence.
2. Needs ≥ 2L terms; works modulo a prime; O(n²).
3. Multipoint: product tree of (x − xᵢ); A mod node product down the tree.
4. Division: reversed quotient = rev(A) · rev(B)⁻¹ mod x^(n−m+1).
